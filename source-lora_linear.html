1"""LoRA: a trainable low-rank correction next to a frozen weight, following Chapter 4 equation by equation.
2 
3Run:   .venv-transformer/bin/python lora_linear.py
4Tests: .venv-transformer/bin/python -m unittest -v test_lora_linear
5 
6Notation of Chapter 2: a projection is Q = X W_Q with W_Q of shape [d, d]; with LoRA it becomes
7Q = X (W_Q + (alpha / r) W_down W_up). PyTorch's nn.Linear stores W_Q transposed, as weight[d_out, d_in],
8which only matters inside merge(). The LoRA paper calls W_down "A" and W_up "B".
9"""
10

22# BEGIN lora-linear
23class LoRALinear(nn.Module):
24    """Q = X W_Q + (alpha / r) (X W_down) W_up, with W_Q frozen and only W_down, W_up trainable."""
25 
26    def __init__(self, base: nn.Linear, r: int, alpha: float):
27        super().__init__()
28        self.base = base
29        for parameter in base.parameters():
30            parameter.requires_grad = False                            # step 2: W_Q (and a bias) frozen
31        d_in, d_out = base.in_features, base.out_features
32        self.scale = alpha / r                                         # alpha / r
33        self.down = nn.Parameter(torch.randn(d_in, r) / math.sqrt(d_in))   # W_down: random, so that W_up gets a gradient
34        self.up = nn.Parameter(torch.zeros(r, d_out))                       # W_up: zero, so that Delta W = 0 at the start
35        self.merged = False
36 
37    def delta_w(self) -> Tensor:
38        return self.scale * self.down @ self.up                        # Delta W: [d, d], rank <= r
39 
40    def forward(self, x: Tensor) -> Tensor:
41        q = self.base(x)                                               # step 2: X W_Q
42        if self.merged:
43            return q
44        return q + self.scale * (x @ self.down) @ self.up              # steps 3-4: X W_down first, then W_up
45 
46    @torch.no_grad()
47    def merge(self) -> None:
48        """Step 8: W_Q <- W_Q + Delta W, once. The layer then costs exactly what a plain Linear costs."""
49        if not self.merged:
50            self.base.weight += self.delta_w().T                       # weight is stored as [d_out, d_in]
51            self.merged = True
52 
53    @torch.no_grad()
54    def unmerge(self) -> None:
55        if self.merged:
56            self.base.weight -= self.delta_w().T
57            self.merged = False
58# END lora-linear

61# BEGIN lora-apply
62def add_lora(model: nn.Module, r: int, alpha: float, names=("w_q", "w_k", "w_v", "w_o")) -> list:
63    """Step 1: freeze the whole model, then wrap every Linear called `names` (W_Q, W_K, W_V, W_O) in a LoRALinear."""
64    for parameter in model.parameters():
65        parameter.requires_grad = False
66    adapters = []
67    for module in list(model.modules()):
68        for name, child in list(module.named_children()):
69            if isinstance(child, nn.Linear) and name in names:
70                adapter = LoRALinear(child, r, alpha)
71                setattr(module, name, adapter)
72                adapters.append(adapter)
73    return adapters
74 
75 
76def merge_all(model: nn.Module) -> None:
77    for module in model.modules():
78        if isinstance(module, LoRALinear):
79            module.merge()
80 
81 
82def count_parameters(model: nn.Module) -> tuple:
83    trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
84    return trainable, sum(p.numel() for p in model.parameters())
85# END lora-apply

89SRC = ["<pad>", "<bos>", "<eos>", "we", "watch", "a", "short", "video", "you"]
90TGT = ["<pad>", "<bos>", "<eos>", "nous", "regardons", "video", "une", "courte", "tu", "regardes"]

93def example(source: str, target: str) -> tuple:
94    """Source ids, teacher-forcing decoder input (BOS + target) and next-token labels (target + EOS)."""
95    s = torch.tensor([[SRC.index(word) for word in source.split()]])
96    full = torch.tensor([[1] + [TGT.index(word) for word in target.split()] + [2]])
97    return s, full[:, :-1], full[:, 1:]
98 
99 
100def train(model: nn.Module, examples: list, steps: int, lr: float = 1e-2) -> float:
101    """Minimize the next-token loss of Chapter 2 over whatever parameters are trainable."""
102    optimizer = torch.optim.Adam([p for p in model.parameters() if p.requires_grad], lr=lr)
103    for _ in range(steps):
104        optimizer.zero_grad()
105        loss = sum(F.cross_entropy(model(s, t).reshape(-1, len(TGT)), y.reshape(-1), ignore_index=0)
106                   for s, t, y in examples) / len(examples)
107        loss.backward()
108        optimizer.step()
109    return loss.item()
110 
111 
112def translate(model: nn.Module, s: Tensor) -> str:
113    return " ".join(TGT[i] for i in greedy_decode(model.eval(), s)[0].tolist()[1:])

116def main() -> None:
117    torch.manual_seed(2)
118    model = Transformer(len(SRC), len(TGT))                             # d = 8, h = 2, d_ff = 32
119    old = [example("we watch a short video", "nous regardons une courte video"),
120           example("you watch video", "tu regardes video")]
121    new = example("we watch video", "nous regardons video")
122 
123    loss = train(model.train(), old, 400)                               # stands in for pretraining
124    print(f"pretrained on two pairs (all parameters), loss {loss:.3f}")
125    for s, _, _ in old + [new]:
126        print(f"  {' '.join(SRC[i] for i in s[0].tolist()):24s} -> {translate(model, s)}")
127 
128    adapters = add_lora(model, r=2, alpha=4.0)                          # step 1: W_Q, W_K, W_V, W_O of every attention
129    frozen = {name: p.clone() for name, p in model.named_parameters() if not p.requires_grad}
130    trainable, total = count_parameters(model)
131    print(f"LoRA r=2 on {len(adapters)} matrices: {trainable} of {total} parameters trainable "
132          f"({100 * trainable / total:.0f}%); each 8x8 matrix gets W_down [8,2] and W_up [2,8]")
133    print(f"  before training Delta W = 0, so the new pair still gives: {translate(model, new[0])}")
134 
135    loss = train(model.train(), [new] + old, 300)                       # steps 5-7: only W_down and W_up move
136    unchanged = all(torch.equal(p, frozen[name]) for name, p in model.named_parameters() if name in frozen)
137    print(f"LoRA trained on the new pair plus the two old ones: loss {loss:.3f}, frozen weights unchanged: {unchanged}")
138    for s, _, _ in [new] + old:
139        print(f"  {' '.join(SRC[i] for i in s[0].tolist()):24s} -> {translate(model, s)}")
140 
141    before = model.eval()(new[0], new[1])
142    merge_all(model)                                                    # step 8
143    print(f"merged: same logits {torch.allclose(before, model(new[0], new[1]), atol=1e-6)}; "
144          f"one attention matrix is again a plain {tuple(adapters[0].base.weight.shape)} weight")
145 
146 
147if __name__ == "__main__":
148    main()
149# END lora-demo
