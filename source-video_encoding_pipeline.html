1"""
2Interview-style video encoding pipeline (encoder only).
3 
4Scope:
5- RGB -> YUV420 preprocess
6- Block split
7- Intra / Inter prediction (simple block matching)
8- Residual transform + quantization
9- Entropy coding (arithmetic coding on syntax tokens)
10- In-loop reconstruction + simple deblock filter
11- Basic rate control over QP
12 
13This is intentionally simplified and not standard-compliant H.264/HEVC/AV1.
14"""

41def rgb_to_yuv420(rgb: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
42    if rgb.ndim != 3 or rgb.shape[2] != 3:
43        raise ValueError("Expected frame shape (H, W, 3)")
44    x = rgb.astype(np.float32)
45    r, g, b = x[..., 0], x[..., 1], x[..., 2]
46    y = 0.299 * r + 0.587 * g + 0.114 * b
47    u = -0.168736 * r - 0.331264 * g + 0.5 * b + 128.0
48    v = 0.5 * r - 0.418688 * g - 0.081312 * b + 128.0
49 
50    h, w = y.shape
51    pad_h, pad_w = h % 2, w % 2
52    if pad_h or pad_w:
53        u = np.pad(u, ((0, pad_h), (0, pad_w)), mode="edge")
54        v = np.pad(v, ((0, pad_h), (0, pad_w)), mode="edge")
55    u420 = 0.25 * (u[0::2, 0::2] + u[0::2, 1::2] + u[1::2, 0::2] + u[1::2, 1::2])
56    v420 = 0.25 * (v[0::2, 0::2] + v[0::2, 1::2] + v[1::2, 0::2] + v[1::2, 1::2])
57    return y.astype(np.float32), u420.astype(np.float32), v420.astype(np.float32)

60def pad_to_block(plane: np.ndarray, block_size: int) -> Tuple[np.ndarray, Tuple[int, int]]:
61    h, w = plane.shape
62    pad_h = (-h) % block_size
63    pad_w = (-w) % block_size
64    if pad_h == 0 and pad_w == 0:
65        return plane, (0, 0)
66    return np.pad(plane, ((0, pad_h), (0, pad_w)), mode="edge"), (pad_h, pad_w)

69def unpad(plane: np.ndarray, pad: Tuple[int, int]) -> np.ndarray:
70    pad_h, pad_w = pad
71    if pad_h:
72        plane = plane[:-pad_h, :]
73    if pad_w:
74        plane = plane[:, :-pad_w]
75    return plane

82def dct_mat_8() -> np.ndarray:
83    global _DCT8
84    if _DCT8 is not None:
85        return _DCT8
86    n = 8
87    c = np.zeros((n, n), dtype=np.float32)
88    for k in range(n):
89        alpha = np.sqrt(1 / n) if k == 0 else np.sqrt(2 / n)
90        for i in range(n):
91            c[k, i] = alpha * np.cos(np.pi * (i + 0.5) * k / n)
92    _DCT8 = c
93    return c

96def dct2(block: np.ndarray) -> np.ndarray:
97    c = dct_mat_8()
98    return c @ block @ c.T

101def idct2(coeff: np.ndarray) -> np.ndarray:
102    c = dct_mat_8()
103    return c.T @ coeff @ c

106def zigzag_indices() -> np.ndarray:
107    global _ZZ
108    if _ZZ is not None:
109        return _ZZ
110    order = []
111    for s in range(15):
112        if s % 2 == 0:
113            for i in range(s + 1):
114                j = s - i
115                if i < 8 and j < 8:
116                    order.append(i * 8 + j)
117        else:
118            for j in range(s + 1):
119                i = s - j
120                if i < 8 and j < 8:
121                    order.append(i * 8 + j)
122    _ZZ = np.array(order, dtype=np.int32)
123    return _ZZ

126def zigzag_scan(block8: np.ndarray) -> np.ndarray:
127    return block8.reshape(-1)[zigzag_indices()]

130def quant_step_from_qp(qp: int) -> float:
131    # Qstep grows roughly exponentially with QP in practical codecs.
132    return float(2 ** ((qp - 4) / 6.0))

135def intra_predict(recon_plane: np.ndarray, by: int, bx: int, block_size: int) -> np.ndarray:
136    if by == 0 and bx == 0:
137        return np.full((block_size, block_size), 128.0, dtype=np.float32)
138    vals = []
139    if bx > 0:
140        vals.append(float(np.mean(recon_plane[by : by + block_size, bx - 1])))
141    if by > 0:
142        vals.append(float(np.mean(recon_plane[by - 1, bx : bx + block_size])))
143    pred_val = float(np.mean(vals)) if vals else 128.0
144    return np.full((block_size, block_size), pred_val, dtype=np.float32)

147def inter_predict(
148    cur_block: np.ndarray,
149    ref_plane: np.ndarray,
150    by: int,
151    bx: int,
152    block_size: int,
153    search_range: int,
154) -> Tuple[np.ndarray, Tuple[int, int]]:
155    h, w = ref_plane.shape
156    y0 = max(0, by - search_range)
157    y1 = min(h - block_size, by + search_range)
158    x0 = max(0, bx - search_range)
159    x1 = min(w - block_size, bx + search_range)
160 
161    best_cost = float("inf")
162    best_pos = (by, bx)
163    for ry in range(y0, y1 + 1):
164        for rx in range(x0, x1 + 1):
165            cand = ref_plane[ry : ry + block_size, rx : rx + block_size]
166            cost = float(np.sum(np.abs(cur_block - cand)))
167            if cost < best_cost:
168                best_cost = cost
169                best_pos = (ry, rx)
170 
171    ry, rx = best_pos
172    pred = ref_plane[ry : ry + block_size, rx : rx + block_size].astype(np.float32)
173    mv = (ry - by, rx - bx)
174    return pred, mv

177def coeffs_to_tokens(zz: np.ndarray) -> List[int]:
178    tokens = [int(zz[0])]  # DC
179    run = 0
180    for v in zz[1:]:
181        value = int(v)
182        if value == 0:
183            run += 1
184            continue
185        tokens.append(run)
186        tokens.append(value)
187        run = 0
188    tokens.append(TOK_EOB)
189    return tokens

192def deblock_filter(plane: np.ndarray, block_size: int) -> np.ndarray:
193    out = plane.copy()
194    h, w = out.shape
195    for x in range(block_size, w, block_size):
196        avg = 0.5 * (out[:, x - 1] + out[:, x])
197        out[:, x - 1] = avg
198        out[:, x] = avg
199    for y in range(block_size, h, block_size):
200        avg = 0.5 * (out[y - 1, :] + out[y, :])
201        out[y - 1, :] = avg
202        out[y, :] = avg
203    return out

206def encode_plane(
207    plane: np.ndarray,
208    ref_plane: Optional[np.ndarray],
209    frame_type: str,
210    qp: int,
211    block_size: int,
212    search_range: int,
213) -> Tuple[np.ndarray, List[int], int]:
214    padded, pad = pad_to_block(plane, block_size)
215    if ref_plane is not None:
216        ref_padded, _ = pad_to_block(ref_plane, block_size)
217    else:
218        ref_padded = None
219 
220    h, w = padded.shape
221    recon = np.zeros((h, w), dtype=np.float32)
222    qstep = quant_step_from_qp(qp)
223    tokens: List[int] = []
224    mv_count = 0
225 
226    for by in range(0, h, block_size):
227        for bx in range(0, w, block_size):
228            cur = padded[by : by + block_size, bx : bx + block_size].astype(np.float32)
229 
230            if frame_type == "P" and ref_padded is not None:
231                pred, mv = inter_predict(cur, ref_padded, by, bx, block_size, search_range)
232                mode = 1
233                mv_count += 1
234            else:
235                pred = intra_predict(recon, by, bx, block_size)
236                mv = (0, 0)
237                mode = 0
238 
239            residual = cur - pred
240            coeff = dct2(residual)
241            qcoeff = np.rint(coeff / qstep).astype(np.int32)
242            zz = zigzag_scan(qcoeff)
243 
244            tokens.extend([TOK_BLOCK_START, mode, int(mv[0]), int(mv[1])])
245            tokens.extend(coeffs_to_tokens(zz))
246 
247            recon_residual = idct2(qcoeff.astype(np.float32) * qstep)
248            recon_block = np.clip(np.rint(pred + recon_residual), 0, 255)
249            recon[by : by + block_size, bx : bx + block_size] = recon_block
250 
251    recon = deblock_filter(recon, block_size)
252    recon = unpad(recon, pad).astype(np.float32)
253    return recon, tokens, mv_count

256def update_qp(prev_qp: int, prev_bits: Optional[int], target_bits_per_frame: float) -> int:
257    if prev_bits is None:
258        return prev_qp
259    qp = prev_qp
260    if prev_bits > target_bits_per_frame * 1.10:
261        qp += 2
262    elif prev_bits < target_bits_per_frame * 0.90:
263        qp -= 2
264    return int(np.clip(qp, 12, 42))

267def encode_frame(
268    frame_rgb: np.ndarray,
269    frame_idx: int,
270    frame_type: str,
271    qp: int,
272    ref_yuv: Optional[Tuple[np.ndarray, np.ndarray, np.ndarray]],
273    block_size: int = 8,
274    search_range: int = 4,
275) -> Tuple[EncodedFrame, Tuple[np.ndarray, np.ndarray, np.ndarray]]:
276    y, u, v = rgb_to_yuv420(frame_rgb)
277    ref_y = ref_u = ref_v = None
278    if ref_yuv is not None:
279        ref_y, ref_u, ref_v = ref_yuv
280 
281    rec_y, tok_y, mv_y = encode_plane(y, ref_y, frame_type, qp, block_size, search_range)
282    rec_u, tok_u, mv_u = encode_plane(u, ref_u, frame_type, qp, block_size, search_range)
283    rec_v, tok_v, mv_v = encode_plane(v, ref_v, frame_type, qp, block_size, search_range)
284 
285    syntax_tokens: List[int] = [TOK_FRAME_Y] + tok_y + [TOK_FRAME_U] + tok_u + [TOK_FRAME_V] + tok_v
286    bitstream, freqs = arithmetic_encode(syntax_tokens)
287    payload = {
288        "bitstream": bitstream,
289        "freqs": freqs,
290        "length": len(syntax_tokens),
291    }
292    bits = len(bitstream)
293 
294    encoded = EncodedFrame(
295        index=frame_idx,
296        frame_type=frame_type,
297        qp=qp,
298        bits=bits,
299        payload=payload,
300        mv_count=(mv_y + mv_u + mv_v),
301    )
302    recon_yuv = (rec_y, rec_u, rec_v)
303    return encoded, recon_yuv

306def encode_video(
307    frames_rgb: Sequence[np.ndarray],
308    gop: int = 8,
309    target_bpp: float = 0.20,
310    start_qp: int = 24,
311    block_size: int = 8,
312    search_range: int = 4,
313) -> Dict[str, object]:
314    if not frames_rgb:
315        return {"frames": [], "total_bits": 0, "avg_bpp": 0.0}
316 
317    h, w = frames_rgb[0].shape[:2]
318    target_bits_per_frame = target_bpp * h * w
319 
320    encoded_frames: List[EncodedFrame] = []
321    prev_bits = None
322    qp = int(start_qp)
323    ref_yuv = None
324 
325    for i, frame in enumerate(frames_rgb):
326        qp = update_qp(qp, prev_bits, target_bits_per_frame)
327        frame_type = "I" if (i % gop == 0) else "P"
328        encoded, recon_yuv = encode_frame(
329            frame_rgb=frame,
330            frame_idx=i,
331            frame_type=frame_type,
332            qp=qp,
333            ref_yuv=ref_yuv,
334            block_size=block_size,
335            search_range=search_range,
336        )
337        encoded_frames.append(encoded)
338        prev_bits = encoded.bits
339        ref_yuv = recon_yuv
340 
341    total_bits = int(sum(f.bits for f in encoded_frames))
342    avg_bpp = float(total_bits / (len(encoded_frames) * h * w))
343    return {"frames": encoded_frames, "total_bits": total_bits, "avg_bpp": avg_bpp}
