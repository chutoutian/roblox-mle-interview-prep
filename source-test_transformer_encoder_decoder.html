33class AttentionTests(unittest.TestCase):

37    def test_matches_a_per_head_loop_written_from_the_equations(self):
38        attention = MultiHeadAttention(8, 2).double()
39        query = torch.randn(2, 3, 8, dtype=torch.float64)
40        key = torch.randn(2, 5, 8, dtype=torch.float64)
41        value = torch.randn(2, 5, 8, dtype=torch.float64)
42        mask = torch.ones(2, 1, 3, 5, dtype=torch.bool)
43        mask[0, :, :, 4] = False   # example 0: the last key is padding
44        mask[1, :, 0, 1:] = False  # example 1: query 0 may read key 0 only
45        actual = attention(query, key, value, mask)
46 
47        # head_i = softmax(Q_i K_i^T / sqrt(d_k)) V_i, written with loops and dot products.
48        q, k, v = attention.w_q(query), attention.w_k(key), attention.w_v(value)
49        expected = torch.zeros(2, 3, 8, dtype=torch.float64)
50        for b in range(2):
51            for j in range(3):
52                for i in range(2):
53                    f = slice(4 * i, 4 * i + 4)  # head i owns projected features 4i .. 4i+3
54                    allowed = [kk for kk in range(5) if mask[b, 0, j, kk]]
55                    scores = torch.stack([q[b, j, f] @ k[b, kk, f] for kk in allowed]) / math.sqrt(4)
56                    weights = torch.softmax(scores, dim=0)
57                    expected[b, j, f] = sum(w * v[b, kk, f] for w, kk in zip(weights, allowed))
58        torch.testing.assert_close(actual, attention.w_o(expected))
59 
60        weights = attention.attention_weights
61        self.assertEqual(tuple(weights.shape), (2, 2, 3, 5))
62        torch.testing.assert_close(weights.sum(dim=-1), torch.ones(2, 2, 3, dtype=torch.float64))
63        self.assertTrue((weights[~mask.expand_as(weights)] == 0).all())

65    def test_matches_torch_scaled_dot_product_attention(self):
66        attention = MultiHeadAttention(8, 2)
67        query, memory = torch.randn(2, 4, 8), torch.randn(2, 5, 8)
68        mask = torch.ones(2, 1, 4, 5, dtype=torch.bool)
69        mask[1, :, :, 3:] = False
70        actual = attention(query, memory, memory, mask)

89    def block_rotation(self, angles) -> torch.Tensor:
90        """The matrix with one 2x2 rotation per feature pair on its diagonal."""
91        R = torch.zeros(2 * len(angles), 2 * len(angles), dtype=torch.float64)
92        for j, angle in enumerate(angles):
93            R[2 * j:2 * j + 2, 2 * j:2 * j + 2] = torch.tensor(
94                [[math.cos(angle), -math.sin(angle)], [math.sin(angle), math.cos(angle)]], dtype=torch.float64)
95        return R

97    def score(self, q, k, q_angle, k_angle) -> float:
98        return (rotate(q, q_angle) @ rotate(k, k_angle).T).item()

303class ScriptedModel(nn.Module):
304    """Fixes the next-token choice so decoding tests never depend on random weights."""
305 
306    pad_id = PAD
307 
308    def __init__(self, predictions):
309        super().__init__()
310        self.predictions = predictions
311        self.encode_calls = 0
312        self.prefixes = []
313        self.grad_enabled = []
314 
315    def source_mask(self, src_ids):
316        return (src_ids != PAD)[:, None, None, :]
317 
318    def encode(self, src_ids):
319        self.encode_calls += 1
320        return src_ids.float()[..., None]
321 
322    def decode(self, ids, memory, src_mask):
323        self.prefixes.append(ids.clone())
324        self.grad_enabled.append(torch.is_grad_enabled())
325        step = min(len(self.prefixes) - 1, len(self.predictions) - 1)
326        logits = torch.full((*ids.shape, 10), -1000.0)
327        for row, token in enumerate(self.predictions[step]):
328            logits[row, -1, token] = 1000.0
329        return logits
