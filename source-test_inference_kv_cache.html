14def tiny(seed: int, num_layers: int = 2) -> CausalLM:
15    torch.manual_seed(seed)
16    return CausalLM(vocab_size=11, d_model=16, num_heads=2, d_ff=32, num_layers=num_layers).double().eval()

19class CacheTests(unittest.TestCase):
20    def test_decode_with_cache_matches_full_recompute(self):
21        model, ids = tiny(0), torch.randint(0, 11, (2, 9))
22        with torch.no_grad():
23            full, _ = model(ids)
24            logits, caches = prefill(model, ids[:, :3])
25            rows = [logits]
26            for t in range(3, 9):
27                logits, caches = decode_step(model, ids[:, t], caches)
28                rows.append(logits)
29        torch.testing.assert_close(torch.stack(rows, dim=1), full[:, 2:], atol=1e-9, rtol=0)
30        self.assertEqual(caches[0]["k"].shape, (2, 2, 9, 8))
31 
32    def test_generate_with_cache_equals_without(self):
33        model, prompt = tiny(1), torch.randint(0, 11, (1, 3))
34        greedy = lambda logits: logits.argmax(dim=-1)
35        self.assertTrue(torch.equal(generate(model, prompt, 8, greedy), generate(model, prompt, 8, greedy, use_cache=False)))
36 
37    def test_llama3_8b_cache_size(self):
38        self.assertEqual(cache_bytes(32, 8, 128, 1), 131072)               # 128 KiB per token
39        self.assertEqual(cache_bytes(32, 8, 128, 8192), 2 ** 30)           # 1 GiB at 8192 tokens
40        self.assertEqual(cache_bytes(32, 32, 128, 1), 4 * cache_bytes(32, 8, 128, 1))  # LLaMA-2-7B: no GQA

43class SamplingTests(unittest.TestCase):
44    def setUp(self):
45        self.z = torch.tensor([[2.0, 1.0, 0.5, 0.0, -1.0]])
46 
47    def test_limits(self):
48        one_hot = torch.tensor([[1.0, 0, 0, 0, 0]])
49        self.assertTrue(torch.equal(filtered_probabilities(self.z, temperature=0), one_hot))
50        self.assertTrue(torch.equal(filtered_probabilities(self.z, top_k=1), one_hot))
51        torch.testing.assert_close(filtered_probabilities(self.z, top_p=1.0), self.z.softmax(-1))
52        torch.testing.assert_close(filtered_probabilities(self.z), self.z.softmax(-1))
53 
54    def test_worked_example(self):
55        torch.testing.assert_close(filtered_probabilities(self.z, temperature=0.5), torch.tensor([[0.829, 0.112, 0.041, 0.015, 0.002]]), atol=1e-3, rtol=0)
56        torch.testing.assert_close(filtered_probabilities(self.z, top_k=2), torch.tensor([[0.731, 0.269, 0, 0, 0]]), atol=1e-3, rtol=0)
57        torch.testing.assert_close(filtered_probabilities(self.z, top_p=0.9), torch.tensor([[0.579, 0.213, 0.129, 0.078, 0]]), atol=1e-3, rtol=0)
58        self.assertEqual(int((filtered_probabilities(self.z, min_p=0.2) > 0).sum()), 3)
59 
60    def test_top_p_keeps_the_crossing_token_and_at_least_one(self):
61        probs = filtered_probabilities(self.z, top_p=0.6)   # 0.563 < 0.6: the second token is needed to reach p
62        self.assertEqual(int((probs > 0).sum()), 2)
63        self.assertEqual(int((filtered_probabilities(self.z, top_p=1e-6) > 0).sum()), 1)
64 
65    def test_repetition_penalty(self):

67        torch.testing.assert_close(penalised, torch.tensor([[1.0, 1.0, 0.5, 0.0, -2.0]]))

70class SampleNextTests(unittest.TestCase):
71    def test_seeded_sampling_is_reproducible_and_respects_truncation(self):
72        z = torch.tensor([[2.0, 1.0, 0.5, 0.0, -1.0]]).repeat(200, 1)
73        first = sample_next(z, top_k=2, generator=torch.Generator().manual_seed(0))
74        second = sample_next(z, top_k=2, generator=torch.Generator().manual_seed(0))
75        self.assertTrue(torch.equal(first, second))
76        self.assertTrue(bool((first <= 1).all()), 'top-k=2 only ever draws the two largest logits')
77        self.assertTrue(torch.equal(sample_next(z, temperature=0), torch.zeros(200, dtype=torch.long)))
