4def motion_search(reference, current,search_range):
5    """
6    Perform motion search between reference and current frames.
7 
8    Args:
9        reference (numpy.ndarray): The reference frame.
10        current (numpy.ndarray): The current frame.
11    
12    
13    Returns:        motion_vectors (numpy.ndarray): The motion vectors for each block.
14    """
15    h = len(current)
16    w = len(current[0])
17 
18    block_size = 16
19    motion_vectors = [[0] * (w // block_size) for _ in range(h // block_size)]
20 
21    pad_h = (block_size - (h % block_size)) % block_size
22    pad_w = (block_size - (w % block_size)) % block_size
23    
24 
25    for i in range(0, h, block_size):
26        for j in range(0, w, block_size):
27            best_mv = (0, 0)
28            min_error = float('inf')
29 
30            for mv_x in range(-search_range, search_range + 1):
31                for mv_y in range(-search_range, search_range + 1):
32                    ref_block_x = j + mv_x
33                    ref_block_y = i + mv_y
34 
35                    if (0 <= ref_block_x < w - block_size) and (0 <= ref_block_y < h - block_size):
36                        ref_block = reference[ref_block_y:ref_block_y + block_size, ref_block_x:ref_block_x + block_size]
37                        curr_block = current[i:i + block_size, j:j + block_size]
38 
39                        error = sum((ref_block - curr_block) ** 2)
40 
41                        if error < min_error:
42                            min_error = error
43                            best_mv = (mv_x, mv_y)
44 
45            motion_vectors[i // block_size][j // block_size] = best_mv
46 
47    return motion_vectors
