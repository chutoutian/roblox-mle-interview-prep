13# dp[i][b]: minimum total distortion for the first i blocks (0..i-1)
14# using exactly b bits.
15# For every candidate (D, R) of block i, if the previous state is
16# dp[i-1][b-R], transition with:
17# dp[i][b] = min(dp[i][b], dp[i-1][b-R] + D).

22def rd_allocate_dp(self, modes: List[List[Tuple[int, int]]], B: int):
23    # modes[i] = [(D, R), ...] for block i
24    n = len(modes)
25    INF = 10**18
26 
27    dp = [[INF] * (B + 1) for _ in range(n + 1)]
28    parent = [[(-1, -1)] * (B + 1) for _ in range(n + 1)]
29    dp[0][0] = 0
30 
31    for i in range(1, n + 1):
32        for b in range(B + 1):
33            for m_idx, (D, R) in enumerate(modes[i - 1]):
34                if b >= R and dp[i - 1][b - R] != INF:
35                    val = dp[i - 1][b - R] + D
36                    if val < dp[i][b]:
37                        dp[i][b] = val
38                        parent[i][b] = (b - R, m_idx)
39 
40    best_b = min(range(B + 1), key=lambda b: dp[n][b])
41    best_D = dp[n][best_b]
42    pick = [0] * n
43    b = best_b
44    for i in range(n, 0, -1):
45        pb, m_idx = parent[i][b]
46        pick[i - 1] = m_idx
47        b = pb
48    return pick, best_D, best_b

77def huffman_encoding(symbols):
78    """
79    Build a Huffman codebook and the corresponding binary tree.
80 
81    Interview-style simplification:
82      - returns only the codebook {symbol -> bitstring}
83      - stores it in a module global so huffman_decoding(bits) can decode without
84        shipping a separate tree/codebook argument.
85    """
86    symbol_count = Counter(symbols)
87    heap = []
88    tie = 0
89    for sym, c in symbol_count.items():
90        # (freq, tie_breaker, node): tie_breaker avoids comparing int vs tuple when freqs tie.
91        heappush(heap, (int(c), tie, int(sym)))
92        tie += 1
93    
94    print("heap:", heap)
95    while len(heap) > 1:
96        f1, _, n1 = heappop(heap)
97        f2, _, n2 = heappop(heap)
98        heappush(heap, (f1 + f2, tie, (n1, n2)))
99        tie += 1
100    
101    print("heap after merge:", heap)
102    def dfs(node, path="", codebook=None):
103        if node is None:
104            return codebook
105        if isinstance(node, int):
106            codebook[node] = path
107            return codebook
108        left, right = node
109        print("left:", left, "right:", right)
110        dfs(left, path + "0", codebook)
111        dfs(right, path + "1", codebook)
112        return codebook

114    root = heap[0][2] if heap else None
115    if isinstance(root, int):
116        # Single-symbol alphabet: ensure a non-empty code so encoding/decoding is well-defined.
117        codebook = {root: "0"}
118    else:
119        codebook = dfs(root, {})

121    global _LAST_CODEBOOK
122    _LAST_CODEBOOK = codebook
123    return codebook

126def huffman_decoding(bits: str) -> List[int]:
127    """
128    Decode a Huffman bitstring back into symbols using the last codebook
129    produced by huffman_encoding().
130 
131    bits: string of '0'/'1'
132    """
133    global _LAST_CODEBOOK
134    if not _LAST_CODEBOOK:
135        return []
136 
137    codebook = _LAST_CODEBOOK
138    rev = {v: k for k, v in codebook.items()}
139 
140    res = []
141    cur = ""
142    for b in bits:
143        cur += b
144        if cur in rev:
145            res.append(rev[cur])
146            cur = ""
147 
148    return res

161def zig_zag_encode(arr):
162    m = len(arr)
163    n = len(arr[0])
164    lay = 0
165    result = []
166    for lay in range(m + n - 1):
167        if lay % 2 == 0:
168            # Even layer: traverse from bottom-left to top-right
169            r = min(lay, m - 1)
170            c = lay - r
171            while r >= 0 and c < n:
172                result.append(arr[r][c])
173                r -= 1
174                c += 1
175        else:
176            # Odd layer: traverse from top-right to bottom-left
177            c = min(lay, n - 1)
178            r = lay - c
179            while c >= 0 and r < m:
180                result.append(arr[r][c])
181                r += 1
182                c -= 1
183    return result

187def zig_zag_decode(arr, m, n):
188    result = [[0] * n for _ in range(m)]
189    index = 0
190    for lay in range(m + n - 1):
191        if lay % 2 == 0:
192            r = min(lay, m - 1)
193            c = lay - r
194            while r >= 0 and c < n:
195                result[r][c] = arr[index]
196                index += 1
197                r -= 1
198                c += 1
199        else:
200            c = min(lay, n - 1)
201            r = lay - c
202            while c >= 0 and r < m:
203                result[r][c] = arr[index]
204                index += 1
205                r += 1
206                c -= 1
207    return result
