1"""
2Simplified, interview-friendly JPEG-like codec (not a standards-compliant .jpg writer).
3 
4Implements the core ideas end-to-end:
5- RGB <-> YCbCr transform (BT.601-ish)
6- 8x8 block DCT / IDCT
7- Quantization / dequantization (quality scaling)
8- Zigzag scan
9- DC differential coding
10- AC run-length coding (EOB / ZRL)
11- Huffman entropy coding
12 
13The output is a custom Python bitstream container that can be decoded by this module.

30def rgb_to_ycbcr(rgb_u8: np.ndarray) -> np.ndarray:
31    """
32    Convert RGB uint8 image to YCbCr float32 image.
33    Shape: (H, W, 3) -> (H, W, 3).
34    """
35    rgb = rgb_u8.astype(np.float32)
36    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
37 
38    # ITU-R BT.601 (full-range-ish). For interview practice, this is fine.
39    y = 0.299 * r + 0.587 * g + 0.114 * b
40    cb = -0.168736 * r - 0.331264 * g + 0.5 * b + 128.0
41    cr = 0.5 * r - 0.418688 * g - 0.081312 * b + 128.0
42    return np.stack([y, cb, cr], axis=-1).astype(np.float32)

45def ycbcr_to_rgb(ycbcr: np.ndarray) -> np.ndarray:
46    """
47    Convert YCbCr float image to RGB uint8 image.
48    Shape: (H, W, 3) -> (H, W, 3).
49    """
50    y = ycbcr[..., 0].astype(np.float32)
51    cb = ycbcr[..., 1].astype(np.float32) - 128.0
52    cr = ycbcr[..., 2].astype(np.float32) - 128.0
53 
54    r = y + 1.402 * cr
55    g = y - 0.344136 * cb - 0.714136 * cr
56    b = y + 1.772 * cb
57    rgb = np.stack([r, g, b], axis=-1)
58    return np.clip(np.rint(rgb), 0, 255).astype(np.uint8)

61def chroma_subsample_420(ycbcr: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
62    """
63    Very simple 4:2:0 subsampling:
64    - Y stays HxW
65    - Cb/Cr are averaged over 2x2 -> (H/2)x(W/2)
66    """
67    y = ycbcr[..., 0]
68    cb = ycbcr[..., 1]
69    cr = ycbcr[..., 2]
70 
71    h, w = y.shape
72    # Pad chroma to even H/W by edge replication (avoids dropping last row/col).
73    pad_h = h % 2
74    pad_w = w % 2
75    if pad_h or pad_w:
76        cb = np.pad(cb, ((0, pad_h), (0, pad_w)), mode="edge")
77        cr = np.pad(cr, ((0, pad_h), (0, pad_w)), mode="edge")
78 
79    cb2 = 0.25 * (cb[0::2, 0::2] + cb[0::2, 1::2] + cb[1::2, 0::2] + cb[1::2, 1::2])
80    cr2 = 0.25 * (cr[0::2, 0::2] + cr[0::2, 1::2] + cr[1::2, 0::2] + cr[1::2, 1::2])
81    return y, cb2.astype(np.float32), cr2.astype(np.float32)

84def chroma_upsample_420(y: np.ndarray, cb2: np.ndarray, cr2: np.ndarray, out_h: int, out_w: int) -> np.ndarray:
85    """
86    Simple nearest-neighbor 4:2:0 upsampling back to HxW.
87    """
88    cb = np.repeat(np.repeat(cb2, 2, axis=0), 2, axis=1)
89    cr = np.repeat(np.repeat(cr2, 2, axis=0), 2, axis=1)
90    cb = cb[:out_h, :out_w]
91    cr = cr[:out_h, :out_w]
92    y = y[:out_h, :out_w]
93    return np.stack([y, cb, cr], axis=-1).astype(np.float32)

101def pad_to_multiple(x: np.ndarray, block: int = 8) -> Tuple[np.ndarray, Tuple[int, int]]:
102    h, w = x.shape
103 
104    ## we want  (h + pad_h ) % block == 0 =>  pad_h = (-h) % block
105    pad_h = (-h) % block
106    pad_w = (-w) % block
107    if pad_h == 0 and pad_w == 0:
108        return x, (0, 0)
109    return np.pad(x, ((0, pad_h), (0, pad_w)), mode="edge"), (pad_h, pad_w)

112def unpad(x: np.ndarray, pad: Tuple[int, int]) -> np.ndarray:
113    pad_h, pad_w = pad
114    if pad_h:
115        x = x[:-pad_h, :]
116    if pad_w:
117        x = x[:, :-pad_w]
118    return x

121def iter_blocks(x: np.ndarray, block: int = 8) -> Iterable[Tuple[int, int, np.ndarray]]:
122    h, w = x.shape
123    for by in range(0, h, block):
124        for bx in range(0, w, block):
125            yield by, bx, x[by : by + block, bx : bx + block]

136def dct_matrix(n: int = 8) -> np.ndarray:
137    global _DCT_MAT_8
138    if n == 8 and _DCT_MAT_8 is not None:
139        return _DCT_MAT_8
140 
141    mat = np.zeros((n, n), dtype=np.float32)
142    for k in range(n):
143        alpha = np.sqrt(1.0 / n) if k == 0 else np.sqrt(2.0 / n)
144        for i in range(n):
145            mat[k, i] = alpha * np.cos((np.pi / n) * (i + 0.5) * k)
146    if n == 8:
147        _DCT_MAT_8 = mat
148    return mat

151def dct2(block8: np.ndarray) -> np.ndarray:
152    c = dct_matrix(8)
153    x = block8.astype(np.float32)
154    return (c @ x @ c.T).astype(np.float32)

158    c = dct_matrix(8)
159    x = coeff8.astype(np.float32)
160    return (c.T @ x @ c).astype(np.float32)

198def quality_scale_table(qtable: np.ndarray, quality: int) -> np.ndarray:
199    """
200    JPEG-like quality scaling. quality in [1,100], 50 means unchanged-ish.
201    """
202    q = int(np.clip(quality, 1, 100))
203    scale = 5000 / q if q < 50 else 200 - 2 * q
204    scaled = np.floor((qtable * scale + 50) / 100)
205    scaled = np.clip(scaled, 1, 255)
206    return scaled.astype(np.float32)

213def zigzag_indices() -> np.ndarray:
214    global _ZZ_IDX, _IZZ_POS
215    if _ZZ_IDX is not None:
216        return _ZZ_IDX
217 
218    idx = []
219    for s in range(15):  # sums 0..14
220        if s % 2 == 0:
221            for i in range(s + 1):
222                j = s - i
223                if i < 8 and j < 8:
224                    idx.append((i, j))
225        else:
226            for j in range(s + 1):
227                i = s - j
228                if i < 8 and j < 8:
229                    idx.append((i, j))
230    _ZZ_IDX = np.array([i * 8 + j for i, j in idx], dtype=np.int32)
231 
232    pos = np.zeros((64,), dtype=np.int32)
233    for k, flat in enumerate(_ZZ_IDX):
234        pos[flat] = k
235    _IZZ_POS = pos
236    return _ZZ_IDX

239def zigzag_scan(block8: np.ndarray) -> np.ndarray:
240    zz = zigzag_indices()
241    flat = block8.reshape(-1)
242    return flat[zz].astype(np.int32)

245def inverse_zigzag(coeffs64: np.ndarray) -> np.ndarray:

265def magnitude_category(v: int) -> int:
266    a = abs(int(v))
267    if a == 0:
268        return 0
269    return int(np.floor(np.log2(a))) + 1

272def amplitude_to_bits(v: int, size: int) -> int:
273    """
274    JPEG "additional bits" mapping.
275    For negative values, bits are in ones-complement form within 'size' bits.
276    """
277    if size == 0:
278        return 0
279    v = int(v)
280    if v >= 0:
281        return v
282    return ((1 << size) - 1) + v
283 
284 
285def bits_to_amplitude(bits: int, size: int) -> int:
286    if size == 0:
287        return 0
288    bits = int(bits)
289    if bits >= (1 << (size - 1)):
290        return bits
291    return bits - ((1 << size) - 1)

294def rle_ac_symbols(ac_coeffs: Sequence[int]) -> List[Tuple[int, int]]:
295    """
296    AC coefficients (63 values) -> list of (RS_symbol, amplitude_value).
297    RS_symbol packs: (run_of_zeros << 4) | size_category.
298    Uses:
299    - EOB: 0x00
300    - ZRL: 0xF0 (16 zeros)
301    """
302    out: List[Tuple[int, int]] = []
303    run = 0
304    for v in ac_coeffs:
305        v = int(v)
306        if v == 0:
307            run += 1
308            continue
309 
310        while run > 15:
311            out.append((0xF0, 0))  # ZRL
312            run -= 16
313 
314        size = magnitude_category(v)
315        rs = (run << 4) | size
316        out.append((rs, v))
317        run = 0
318 
319    # Only emit EOB when there are remaining trailing zeros (i.e., last coefficients are 0).
320    # If the final AC coefficient is non-zero, the block is "fully specified" and EOB is redundant.
321    if run > 0:
322        out.append((0x00, 0))  # EOB
323    return out

331class BitWriter:
332    def __init__(self) -> None:
333        self._buf = bytearray()
334        self._cur = 0
335        self._nbits = 0
336        self._bitlen = 0
337 
338    @property
339    def bit_length(self) -> int:
340        return self._bitlen
341 
342    def write_bits(self, bits: int, n: int) -> None:
343        if n == 0:
344            return
345        bits &= (1 << n) - 1
346        self._bitlen += n
347        while n > 0:
348            take = min(8 - self._nbits, n)
349            shift = n - take
350            chunk = (bits >> shift) & ((1 << take) - 1)
351            self._cur = (self._cur << take) | chunk
352            self._nbits += take
353            n -= take
354            if self._nbits == 8:
355                self._buf.append(self._cur & 0xFF)
356                self._cur = 0
357                self._nbits = 0
358 
359    def finish(self) -> bytes:
360        if self._nbits:
361            self._cur <<= 8 - self._nbits
362            self._buf.append(self._cur & 0xFF)
363            self._cur = 0
364            self._nbits = 0
365        return bytes(self._buf)
366

368class BitReader:
369    def __init__(self, data: bytes, bit_length: int) -> None:
370        self._data = data
371        self._bit_length = int(bit_length)
372        self._pos = 0  # bits read
373        self._byte_i = 0
374        self._cur = 0
375        self._nbits = 0
376 
377    def read_bits(self, n: int) -> int:
378        if n == 0:
379            return 0
380        if self._pos + n > self._bit_length:
381            raise ValueError("Bitstream underflow")
382        out = 0
383        while n > 0:
384            if self._nbits == 0:
385                self._cur = self._data[self._byte_i]
386                self._byte_i += 1
387                self._nbits = 8
388            take = min(self._nbits, n)
389            shift = self._nbits - take
390            chunk = (self._cur >> shift) & ((1 << take) - 1)
391            out = (out << take) | chunk
392            self._nbits -= take
393            self._cur &= (1 << shift) - 1 if shift else 0
394            n -= take
395            self._pos += take
396        return out

404@dataclass(frozen=True)
405class HuffmanTable:
406    # symbol -> (code_bits, code_length)
407    codes: Dict[int, Tuple[int, int]]
408 
409    def build_decoder(self) -> "HuffmanDecoder":
410        return HuffmanDecoder.from_codes(self.codes)
411 
412 
413class HuffmanDecoder:
414    def __init__(self) -> None:
415        self._next: Dict[int, "HuffmanDecoder"] = {}
416        self._symbol: Optional[int] = None
417 
418    @staticmethod
419    def from_codes(codes: Dict[int, Tuple[int, int]]) -> "HuffmanDecoder":
420        root = HuffmanDecoder()
421        for sym, (bits, n) in codes.items():
422            node = root
423            for i in range(n - 1, -1, -1):
424                b = (bits >> i) & 1
425                if b not in node._next:
426                    node._next[b] = HuffmanDecoder()
427                node = node._next[b]
428            node._symbol = int(sym)
429        return root
430 
431    def decode_symbol(self, br: BitReader) -> int:
432        node = self
433        while node._symbol is None:
434            b = br.read_bits(1)
435            if b not in node._next:
436                raise ValueError("Invalid Huffman code")
437            node = node._next[b]
438        return int(node._symbol)

441def _build_huffman_codes(symbol_counts: Dict[int, int]) -> Dict[int, Tuple[int, int]]:
442    """
443    Build Huffman codes using a classic heap-based tree.
444    Returns symbol -> (bits, length).
445    """
446    # Ensure non-empty: add a dummy symbol if needed.
447    if not symbol_counts:
448        symbol_counts = {0: 1}
449 
450    # Heap nodes: (freq, tie, node)
451    # node: int (symbol) or (left, right)
452    heap: List[Tuple[int, int, object]] = []
453    tie = 0
454    for sym, c in symbol_counts.items():
455        heappush(heap, (int(c), tie, int(sym)))
456        tie += 1
457 
458    if len(heap) == 1:
459        _, _, only = heap[0]
460        return {int(only): (0b0, 1)}
461 
462    while len(heap) > 1:
463        f1, _, n1 = heappop(heap)
464        f2, _, n2 = heappop(heap)
465        heappush(heap, (f1 + f2, tie, (n1, n2)))
466        tie += 1
467 
468    _, _, root = heap[0]
469    codes: Dict[int, Tuple[int, int]] = {}
470 
471    def walk(node: object, bits: int, n: int) -> None:
472        if isinstance(node, int):
473            codes[int(node)] = (bits, n)
474            return
475        left, right = node  # type: ignore[misc]
476        walk(left, (bits << 1) | 0, n + 1)
477        walk(right, (bits << 1) | 1, n + 1)
478 
479    walk(root, 0, 0)
480 
481    # Guard: no zero-length code.
482    for sym, (bits, n) in list(codes.items()):
483        if n == 0:
484            codes[sym] = (0, 1)
485    return codes
486 
487 
488def build_huffman_table(symbols: Iterable[int]) -> HuffmanTable:
489    counts: Dict[int, int] = {}
490    for s in symbols:
491        counts[int(s)] = counts.get(int(s), 0) + 1
492    return HuffmanTable(codes=_build_huffman_codes(counts))

527def _encode_channel_symbols(
528    channel: np.ndarray, qtable: np.ndarray
529) -> Tuple[List[Tuple[int, int, List[Tuple[int, int]]]], List[int], List[int]]:
530    """
531    Returns:
532    - per-block symbols: [(dc_size, dc_val, [(rs, ac_val), ...]), ...]
533    - dc_size_symbols (for Huffman table building)
534    - ac_rs_symbols (for Huffman table building)
535    """
536    blocks: List[Tuple[int, int, List[Tuple[int, int]]]] = []
537    dc_sizes: List[int] = []
538    ac_rs: List[int] = []
539 
540    prev_dc = 0
541    for _, _, b in iter_blocks(channel, 8):
542        shifted = b.astype(np.float32) - 128.0
543        coeff = dct2(shifted)
544        qcoeff = quantize(coeff, qtable)
545        zz = zigzag_scan(qcoeff)
546 
547        dc = int(zz[0])
548        dcdiff = dc - prev_dc
549        prev_dc = dc
550        dc_size = magnitude_category(dcdiff)
551        dc_sizes.append(dc_size)
552 
553        ac_pairs = rle_ac_symbols(zz[1:].tolist())
554        ac_rs.extend([rs for rs, _ in ac_pairs])
555 
556        blocks.append((dc_size, dcdiff, ac_pairs))
557 
558    return blocks, dc_sizes, ac_rs

561def _write_channel_bits(
562    bw: BitWriter,
563    blocks: Sequence[Tuple[int, int, List[Tuple[int, int]]]],
564    dc_table: HuffmanTable,
565    ac_table: HuffmanTable,
566) -> None:
567    dc_codes = dc_table.codes
568    ac_codes = ac_table.codes
569    for dc_size, dc_val, ac_pairs in blocks:
570        bits, n = dc_codes[int(dc_size)]
571        bw.write_bits(bits, n)
572        bw.write_bits(amplitude_to_bits(dc_val, dc_size), dc_size)
573 
574        for rs, v in ac_pairs:
575            bits, n = ac_codes[int(rs)]
576            bw.write_bits(bits, n)
577            size = int(rs) & 0x0F
578            bw.write_bits(amplitude_to_bits(v, size), size)

581def jpeg_like_encode(rgb_u8: np.ndarray, quality: int = 50, subsampling: str = "420") -> JPEGLikeBitstream:
582    """
583    Encode an (H,W,3) uint8 RGB image into a simplified JPEG-like bitstream.
584    subsampling: "444" or "420"
585    """
586    if rgb_u8.ndim != 3 or rgb_u8.shape[2] != 3:
587        raise ValueError("Expected input shape (H, W, 3)")
588    if rgb_u8.dtype != np.uint8:
589        raise ValueError("Expected uint8 RGB input")
590    if subsampling not in ("444", "420"):
591        raise ValueError("subsampling must be '444' or '420'")
592    ### 444: no subsampling
593    ### 422: H x (W/2)
594    h, w, _ = rgb_u8.shape
595    ycbcr = rgb_to_ycbcr(rgb_u8)
596 
597    if subsampling == "420": # average 2x2
598        y, cb, cr = chroma_subsample_420(ycbcr)
599    else:
600        y = ycbcr[..., 0]
601        cb = ycbcr[..., 1]
602        cr = ycbcr[..., 2]
603 
604    y, y_pad = pad_to_multiple(y, 8)
605    cb, cb_pad = pad_to_multiple(cb, 8)
606    cr, cr_pad = pad_to_multiple(cr, 8)
607 
608    q_luma = quality_scale_table(LUMA_Q, quality)
609    q_chroma = quality_scale_table(CHROMA_Q, quality)
610 
611    y_blocks, y_dc_sizes, y_ac_rs = _encode_channel_symbols(y, q_luma)
612    cb_blocks, cb_dc_sizes, cb_ac_rs = _encode_channel_symbols(cb, q_chroma)
613    cr_blocks, cr_dc_sizes, cr_ac_rs = _encode_channel_symbols(cr, q_chroma)
614 
615    dc_luma = build_huffman_table(y_dc_sizes)
616    ac_luma = build_huffman_table(y_ac_rs)
617    dc_chroma = build_huffman_table(cb_dc_sizes + cr_dc_sizes)
618    ac_chroma = build_huffman_table(cb_ac_rs + cr_ac_rs)
619 
620    bw = BitWriter()
621    _write_channel_bits(bw, y_blocks, dc_luma, ac_luma)
622    _write_channel_bits(bw, cb_blocks, dc_chroma, ac_chroma)
623    _write_channel_bits(bw, cr_blocks, dc_chroma, ac_chroma)
624    data = bw.finish()
625 
626    return JPEGLikeBitstream(
627        height=h,
628        width=w,
629        subsampling=subsampling,
630        quality=int(np.clip(quality, 1, 100)),
631        y_pad=y_pad,
632        cb_pad=cb_pad,
633        cr_pad=cr_pad,
634        y_shape_padded=tuple(int(x) for x in y.shape),
635        cb_shape_padded=tuple(int(x) for x in cb.shape),
636        cr_shape_padded=tuple(int(x) for x in cr.shape),
637        q_luma=q_luma.astype(np.float32),
638        q_chroma=q_chroma.astype(np.float32),
639        dc_luma=dc_luma,
640        ac_luma=ac_luma,
641        dc_chroma=dc_chroma,
642        ac_chroma=ac_chroma,
643        data=data,
644        bit_length=bw.bit_length,
645    )

648def _decode_channel(
649    br: BitReader,
650    shape_padded: Tuple[int, int],
651    pad: Tuple[int, int],
652    qtable: np.ndarray,
653    dc_table: HuffmanTable,
654    ac_table: HuffmanTable,
655) -> np.ndarray:
656    h, w = shape_padded
657    out = np.zeros((h, w), dtype=np.float32)
658 
659    dc_dec = dc_table.build_decoder()
660    ac_dec = ac_table.build_decoder()
661 
662    prev_dc = 0
663    for by in range(0, h, 8):
664        for bx in range(0, w, 8):
665            dc_size = dc_dec.decode_symbol(br)
666            dc_bits = br.read_bits(dc_size)
667            dcdiff = bits_to_amplitude(dc_bits, dc_size)
668            dc = prev_dc + dcdiff
669            prev_dc = dc
670 
671            zz = np.zeros((64,), dtype=np.int32)
672            zz[0] = int(dc)
673 
674            k = 1
675            while k < 64:
676                rs = ac_dec.decode_symbol(br)
677                if rs == 0x00:
678                    break  # EOB
679                if rs == 0xF0:
680                    k += 16
681                    continue
682 
683                run = (rs >> 4) & 0x0F
684                size = rs & 0x0F
685                k += run
686                if k >= 64:
687                    raise ValueError("Invalid RLE (overrun)")
688                amp_bits = br.read_bits(size)
689                v = bits_to_amplitude(amp_bits, size)
690                zz[k] = int(v)
691                k += 1
692 
693            qcoeff = inverse_zigzag(zz)
694            coeff = dequantize(qcoeff, qtable)
695            block = idct2(coeff) + 128.0
696            put_block(out, by, bx, block)
697 
698    out = np.clip(out, 0, 255)
699    out = unpad(out, pad)
700    return out

703def jpeg_like_decode(stream: JPEGLikeBitstream) -> np.ndarray:
704    """
705    Decode a JPEGLikeBitstream back to an (H,W,3) uint8 RGB image.
706    """
707    br = BitReader(stream.data, stream.bit_length)
708 
709    y = _decode_channel(
710        br,
711        stream.y_shape_padded,
712        stream.y_pad,
713        stream.q_luma,
714        stream.dc_luma,
715        stream.ac_luma,
716    )
717    cb = _decode_channel(
718        br,
719        stream.cb_shape_padded,
720        stream.cb_pad,
721        stream.q_chroma,
722        stream.dc_chroma,
723        stream.ac_chroma,
724    )
725    cr = _decode_channel(
726        br,
727        stream.cr_shape_padded,
728        stream.cr_pad,
729        stream.q_chroma,
730        stream.dc_chroma,
731        stream.ac_chroma,
732    )
733 
734    if stream.subsampling == "420":
735        ycbcr = chroma_upsample_420(y, cb, cr, stream.height, stream.width)
736    else:
737        ycbcr = np.stack([y, cb, cr], axis=-1).astype(np.float32)
738        ycbcr = ycbcr[: stream.height, : stream.width, :]
739 
740    return ycbcr_to_rgb(ycbcr)

748def mse(a: np.ndarray, b: np.ndarray) -> float:
749    a = a.astype(np.float32)
750    b = b.astype(np.float32)
751    return float(np.mean((a - b) ** 2))
752

754def psnr(a: np.ndarray, b: np.ndarray, max_val: float = 255.0) -> float:
755    m = mse(a, b)
756    if m == 0:
757        return float("inf")
758    return float(20.0 * np.log10(max_val) - 10.0 * np.log10(m))
