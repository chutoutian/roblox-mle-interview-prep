1"""Vision transformers: the mechanisms of Chapter 2, "Vision Transformers, from ViT to today".
2 
3One small module per idea, each written to match its equation in the book: ViT and DeiT
4(class, distillation and register tokens), Swin (windows, shifted windows, relative position
5bias, patch merging), PVT (spatial-reduction attention), MAE (random masking), DINO
6(self-distillation loss, EMA teacher), TimeSformer and ViViT (tubelets, divided space-time
7attention), NaViT (packed sequences) and DiT (adaLN-Zero).
8Run:   .venv-transformer/bin/python vision_transformers.py
9Tests: .venv-transformer/bin/python -m unittest -v test_vision_transformers
10 
11Tokens are batch-first, [B, N, D]; images are channel-first, [B, C, H, W]. Masks are boolean
12and True means "this key may be read", as in Chapter 2.
13"""

27class Attention(nn.Module):
28    """head_i = softmax(Q_i K_i^T / sqrt(d_k) + bias, masked) V_i;  out = [head_1 || ... || head_h] W_O.
29 
30    Chapter 2's attention plus an additive bias (Swin's relative positions). The queries come
31    from x; keys and values from `context`, which is x itself unless given (PVT's pooled keys).
32    """
33 
34    def __init__(self, dim: int, heads: int):
35        super().__init__()
36        self.h, self.d_k = heads, dim // heads
37        self.q = nn.Linear(dim, dim)              # W_Q for all heads
38        self.kv = nn.Linear(dim, 2 * dim)         # W_K and W_V side by side
39        self.out = nn.Linear(dim, dim)            # W_O
40 
41    def forward(self, x: Tensor, context: Tensor = None, bias: Tensor = None, mask: Tensor = None) -> Tensor:
42        context = x if context is None else context
43        B, Lq, D = x.shape
44        Lk = context.shape[1]
45        q = self.q(x).view(B, Lq, self.h, self.d_k).transpose(1, 2)                   # [B, h, Lq, d_k]
46        k, v = self.kv(context).view(B, Lk, 2, self.h, self.d_k).permute(2, 0, 3, 1, 4)  # 2 x [B, h, Lk, d_k]
47        scores = q @ k.transpose(-2, -1) / math.sqrt(self.d_k)                         # [B, h, Lq, Lk]
48        if bias is not None:
49            scores = scores + bias                                                     # broadcasts over B
50        if mask is not None:
51            scores = scores.masked_fill(~mask, float("-inf"))
52        heads = scores.softmax(dim=-1) @ v                                             # [B, h, Lq, d_k]
53        return self.out(heads.transpose(1, 2).reshape(B, Lq, D))

56def mlp(dim: int, ratio: int = 4) -> nn.Sequential:
57    return nn.Sequential(nn.Linear(dim, ratio * dim), nn.GELU(), nn.Linear(ratio * dim, dim))

60class Block(nn.Module):
61    """Pre-norm encoder block (Chapter 3, step 3):  x' = x + MSA(LN(x));  x = x' + MLP(LN(x'))."""
62 
63    def __init__(self, dim: int, heads: int):
64        super().__init__()
65        self.norm1, self.attn = nn.LayerNorm(dim), Attention(dim, heads)
66        self.norm2, self.mlp = nn.LayerNorm(dim), mlp(dim)
67 
68    def forward(self, x: Tensor, mask: Tensor = None) -> Tensor:
69        x = x + self.attn(self.norm1(x), mask=mask)
70        return x + self.mlp(self.norm2(x))

75class PatchEmbed(nn.Module):
76    """x_p^i E for every patch at once. Conv2d with kernel = stride = P equals the Linear layer
77    applied to the flattened patches of patchify(); the output is in row-major patch order."""
78 
79    def __init__(self, patch: int, channels: int, dim: int):
80        super().__init__()
81        self.patch = patch
82        self.proj = nn.Conv2d(channels, dim, kernel_size=patch, stride=patch)
83 
84    def forward(self, images: Tensor) -> Tensor:
85        return self.proj(images).flatten(2).transpose(1, 2)          # [B, C, H, W] -> [B, N, D]

88def resize_positions(positions: Tensor, lead: int, grid: tuple) -> Tensor:
89    """Another resolution at fine-tuning: 2D interpolation of the patch rows of E_pos. Bicubic,
90    as in DeiT, which notes that bilinear shrinks the vectors; the ViT code interpolates linearly.
91 
92    positions: [1, lead + n*n, D]. The `lead` rows (class and distillation tokens) are kept.
93    """
94    head, table = positions[:, :lead], positions[:, lead:]
95    side = math.isqrt(table.shape[1])
96    if (side, side) == tuple(grid):
97        return positions
98    D = table.shape[-1]
99    table = table.reshape(1, side, side, D).permute(0, 3, 1, 2)                  # [1, D, n, n]
100    table = F.interpolate(table, size=tuple(grid), mode="bicubic", align_corners=False)
101    return torch.cat([head, table.permute(0, 2, 3, 1).reshape(1, -1, D)], dim=1)

104class ViT(nn.Module):
105    """z0 = [x_cls; (x_dist;) x_p^1 E; ...; x_p^N E] + E_pos  ->  L blocks  ->  LN(z_L^0) W_head.
106 
107    distill=True is DeiT: a second learned token with its own head, trained on the teacher's
108    labels. registers=r adds r learned tokens without positions that join attention and are
109    dropped at the output (Vision Transformers Need Registers).
110    """
111 
112    def __init__(self, image_size: int = 224, patch: int = 16, channels: int = 3, dim: int = 768,
113                 depth: int = 12, heads: int = 12, classes: int = 1000, distill: bool = False,
114                 registers: int = 0):
115        super().__init__()
116        self.embed = PatchEmbed(patch, channels, dim)
117        self.lead = 1 + int(distill)                                              # tokens before the patches
118        self.cls = nn.Parameter(torch.zeros(1, 1, dim))                           # x_cls
119        self.dist = nn.Parameter(torch.zeros(1, 1, dim)) if distill else None     # x_dist (DeiT)
120        self.registers = nn.Parameter(torch.zeros(1, registers, dim)) if registers else None
121        grid = image_size // patch
122        self.positions = nn.Parameter(0.02 * torch.randn(1, self.lead + grid * grid, dim))  # E_pos
123        self.blocks = nn.ModuleList(Block(dim, heads) for _ in range(depth))
124        self.norm = nn.LayerNorm(dim)
125        self.head = nn.Linear(dim, classes)                                       # W_head
126        self.head_dist = nn.Linear(dim, classes) if distill else None
127 
128    def forward(self, images: Tensor):
129        B, _, H, W = images.shape
130        P = self.embed.patch
131        x = self.embed(images)                                                    # [B, N, D]
132        lead = [self.cls.expand(B, -1, -1)]
133        if self.dist is not None:
134            lead.append(self.dist.expand(B, -1, -1))
135        x = torch.cat(lead + [x], dim=1) + resize_positions(self.positions, self.lead, (H // P, W // P))
136        if self.registers is not None:
137            r = self.registers.expand(B, -1, -1)
138            x = torch.cat([x[:, :self.lead], r, x[:, self.lead:]], dim=1)
139        for block in self.blocks:
140            x = block(x)                                                          # z_1 ... z_L
141        x = self.norm(x)
142        if self.head_dist is None:
143            return self.head(x[:, 0])                                             # [B, K] from z_L^0
144        return self.head(x[:, 0]), self.head_dist(x[:, 1])                        # class and distillation heads

147def deit_loss(logits_cls: Tensor, logits_dist: Tensor, labels: Tensor, teacher_logits: Tensor) -> Tensor:
148    """Hard distillation:  L = 1/2 CE(psi(Z_cls), y) + 1/2 CE(psi(Z_dist), y_t),  y_t = argmax_c Z_t(c)."""
149    y_t = teacher_logits.argmax(dim=-1)
150    return 0.5 * F.cross_entropy(logits_cls, labels) + 0.5 * F.cross_entropy(logits_dist, y_t)
151 
152 
153def deit_predict(model: ViT, images: Tensor) -> Tensor:
154    """At test time the paper adds the softmax outputs of the two heads (here: their mean, which
155    ranks the classes the same way); the official code averages the logits instead."""
156    logits_cls, logits_dist = model(images)
157    return (logits_cls.softmax(dim=-1) + logits_dist.softmax(dim=-1)) / 2

162def window_partition(x: Tensor, M: int) -> Tensor:
163    """[B, H, W, C] -> [B * (H/M) * (W/M), M*M, C]: each M x M window becomes one short sequence."""
164    B, H, W, C = x.shape
165    x = x.view(B, H // M, M, W // M, M, C).permute(0, 1, 3, 2, 4, 5)
166    return x.reshape(-1, M * M, C)
167 
168 
169def window_reverse(windows: Tensor, M: int, H: int, W: int) -> Tensor:
170    """Inverse of window_partition: [B * (H/M) * (W/M), M*M, C] -> [B, H, W, C]."""
171    C = windows.shape[-1]
172    x = windows.view(-1, H // M, W // M, M, M, C).permute(0, 1, 3, 2, 4, 5)
173    return x.reshape(-1, H, W, C)

176def relative_position_index(M: int) -> Tensor:
177    """[M*M, M*M] indices into the flattened (2M-1) x (2M-1) table:
178    B_ij = B_hat[y_i - y_j + M - 1, x_i - x_j + M - 1]."""
179    ys, xs = torch.meshgrid(torch.arange(M), torch.arange(M), indexing="ij")
180    y, x = ys.flatten(), xs.flatten()                                  # row-major order inside a window
181    dy = y[:, None] - y[None, :] + M - 1                               # 0 ... 2M-2
182    dx = x[:, None] - x[None, :] + M - 1
183    return dy * (2 * M - 1) + dx

186def shifted_window_mask(H: int, W: int, M: int, s: int) -> Tensor:
187    """[windows, M*M, M*M], True = may read. After the roll by -s, a window can hold pieces of
188    up to four regions that were not neighbours in the image; only same-region pairs attend.
189    The official code adds -100 to the other pairs (the book's mask); Attention here uses -inf,
190    which gives exact zeros after the softmax."""
191    region = torch.zeros(1, H, W, 1, dtype=torch.long)
192    label = 0
193    for rows in (slice(0, -M), slice(-M, -s), slice(-s, None)):
194        for cols in (slice(0, -M), slice(-M, -s), slice(-s, None)):
195            region[:, rows, cols] = label
196            label += 1
197    r = window_partition(region, M).squeeze(-1)                         # [windows, M*M]
198    return r[:, :, None] == r[:, None, :]

201class SwinBlock(nn.Module):
202    """z_hat = (S)W-MSA(LN(z)) + z;  z = MLP(LN(z_hat)) + z_hat.  shift = 0 is W-MSA, M // 2 is SW-MSA."""
203 
204    def __init__(self, dim: int, heads: int, M: int = 7, shift: int = 0):
205        super().__init__()
206        self.M, self.shift = M, shift
207        self.norm1, self.attn = nn.LayerNorm(dim), Attention(dim, heads)
208        self.table = nn.Parameter(0.02 * torch.randn(heads, (2 * M - 1) ** 2))     # B_hat, one per head
209        self.register_buffer("index", relative_position_index(M), persistent=False)
210        self.norm2, self.mlp = nn.LayerNorm(dim), mlp(dim)
211 
212    def mix(self, h: Tensor, H: int, W: int) -> Tensor:
213        """The attention sublayer on a normalized map h: [B, H*W, C] -> [B, H*W, C]."""
214        B, N, C = h.shape
215        M = self.M
216        s = self.shift if min(H, W) > M else 0                         # one window covers the map: no shift
217        h = h.view(B, H, W, C)
218        if s:
219            h = torch.roll(h, shifts=(-s, -s), dims=(1, 2))            # cyclic shift up and left
220        windows = window_partition(h, M)                               # [B * windows, M*M, C]
221        bias = self.table[:, self.index]                               # B: [heads, M*M, M*M]
222        mask = None
223        if s:
224            mask = shifted_window_mask(H, W, M, s).to(h.device).repeat(B, 1, 1)[:, None]
225        h = window_reverse(self.attn(windows, bias=bias, mask=mask), M, H, W)
226        if s:
227            h = torch.roll(h, shifts=(s, s), dims=(1, 2))              # undo the shift
228        return h.reshape(B, N, C)
229 
230    def forward(self, x: Tensor, H: int, W: int) -> Tensor:
231        x = x + self.mix(self.norm1(x), H, W)
232        return x + self.mlp(self.norm2(x))

235class PatchMerging(nn.Module):
236    """x'_{u,v} = LN([x_{2u,2v}; x_{2u+1,2v}; x_{2u,2v+1}; x_{2u+1,2v+1}]) W_merge:
237    [B, H*W, C] -> [B, (H/2)*(W/2), 2C]. Half the grid, twice the width."""
238 
239    def __init__(self, dim: int):
240        super().__init__()
241        self.norm = nn.LayerNorm(4 * dim)
242        self.reduction = nn.Linear(4 * dim, 2 * dim, bias=False)       # W_merge
243 
244    def forward(self, x: Tensor, H: int, W: int) -> Tensor:
245        B, _, C = x.shape
246        x = x.view(B, H, W, C)
247        x = torch.cat([x[:, 0::2, 0::2], x[:, 1::2, 0::2], x[:, 0::2, 1::2], x[:, 1::2, 1::2]], dim=-1)
248        return self.reduction(self.norm(x.reshape(B, -1, 4 * C)))

251class Swin(nn.Module):
252    """4 x 4 patches -> C, then four stages of (W-MSA, SW-MSA) block pairs with merging in between.
253    Returns the four feature maps, at strides 4, 8, 16 and 32 like a ResNet."""
254 
255    def __init__(self, channels: int = 3, dim: int = 96, depths=(2, 2, 6, 2), heads=(3, 6, 12, 24),
256                 M: int = 7, patch: int = 4):
257        super().__init__()
258        self.embed = PatchEmbed(patch, channels, dim)                  # 4*4*3 = 48 numbers -> C
259        self.stages = nn.ModuleList(
260            nn.ModuleList(SwinBlock(dim * 2 ** i, h, M, shift=0 if j % 2 == 0 else M // 2) for j in range(n))
261            for i, (n, h) in enumerate(zip(depths, heads)))
262        self.merges = nn.ModuleList(PatchMerging(dim * 2 ** i) for i in range(len(depths) - 1))
263 
264    def forward(self, images: Tensor) -> list:
265        x = self.embed(images)
266        H, W = images.shape[2] // self.embed.patch, images.shape[3] // self.embed.patch
267        features = []
268        for i, blocks in enumerate(self.stages):
269            for block in blocks:
270                x = block(x, H, W)
271            features.append(x)                                         # [B, H*W, C * 2^i]
272            if i < len(self.merges):
273                x, H, W = self.merges[i](x, H, W), H // 2, W // 2
274        return features
275# END vit-swin

278# BEGIN vit-sra
279class SpatialReductionAttention(nn.Module):
280    """PVT:  SRA(Q, K, V) = Attn(Q W_Q, SR(K) W_K, SR(V) W_V),  SR(x) = LN(Reshape(x, R) W_S).
281 
282    Reshape(x, R) turns every R x R block of tokens into one row of R*R*C numbers and W_S maps it
283    back to C, so N queries read N / R^2 keys. Conv2d(kernel = stride = R) is exactly that.
284    """
285 
286    def __init__(self, dim: int, heads: int, R: int):
287        super().__init__()
288        self.R = R
289        self.attn = Attention(dim, heads)
290        if R > 1:
291            self.sr = nn.Conv2d(dim, dim, kernel_size=R, stride=R)     # Reshape(x, R) W_S
292            self.norm = nn.LayerNorm(dim)
293 
294    def forward(self, x: Tensor, H: int, W: int) -> Tensor:
295        if self.R == 1:
296            return self.attn(x)
297        B, N, C = x.shape
298        reduced = self.sr(x.transpose(1, 2).reshape(B, C, H, W)).flatten(2).transpose(1, 2)  # [B, N/R^2, C]
299        return self.attn(x, context=self.norm(reduced))
300# END vit-sra

303# BEGIN vit-mae
304def random_masking(x: Tensor, ratio: float, generator: torch.Generator = None):
305    """Keep a random (1 - ratio) of the N tokens of every image, sampled without replacement.
306 
307    Returns the kept tokens [B, N_keep, D], the mask [B, N] in image order (1 = removed) and
308    ids_restore [B, N], the permutation that puts a shuffled sequence back in image order.
309    """
310    B, N, D = x.shape
311    keep = int(N * (1 - ratio))
312    noise = torch.rand(B, N, generator=generator).to(x.device)
313    ids_shuffle = noise.argsort(dim=1)                                 # the smallest noise is kept
314    ids_restore = ids_shuffle.argsort(dim=1)
315    kept = torch.gather(x, 1, ids_shuffle[:, :keep, None].expand(-1, -1, D))
316    mask = torch.ones(B, N, device=x.device)
317    mask[:, :keep] = 0
318    return kept, torch.gather(mask, 1, ids_restore), ids_restore
319 
320 
321def unshuffle(h: Tensor, mask_token: Tensor, ids_restore: Tensor) -> Tensor:
322    """Decoder input: the encoded visible tokens plus the shared mask token m for every removed
323    patch, back in image order. [B, N_keep, D] -> [B, N, D]."""
324    B, keep, D = h.shape
325    N = ids_restore.shape[1]
326    full = torch.cat([h, mask_token.expand(B, N - keep, D)], dim=1)   # still in shuffled order
327    return torch.gather(full, 1, ids_restore[:, :, None].expand(-1, -1, D))

330def mae_loss(pred: Tensor, patches: Tensor, mask: Tensor) -> Tensor:
331    """L = mean over removed patches j of ||x_hat_j - norm(x_j)||^2 / (P*P*C), with per-patch
332    normalized pixels as the target. Visible patches do not count."""
333    mean, var = patches.mean(dim=-1, keepdim=True), patches.var(dim=-1, keepdim=True)
334    target = (patches - mean) / (var + 1e-6) ** 0.5
335    per_patch = ((pred - target) ** 2).mean(dim=-1)                    # [B, N]
336    return (per_patch * mask).sum() / mask.sum()

339class MAE(nn.Module):
340    """Encoder on the visible quarter only, a small decoder on all N positions, pixels out.
341    Defaults: a ViT-B encoder (the paper's ablations use ViT-L/16) with the paper's 8-block,
342    512-wide decoder, 75% masked. The paper also carries a class token and uses fixed 2D
343    sine-cosine positions; this sketch learns them."""
344 
345    def __init__(self, image_size: int = 224, patch: int = 16, channels: int = 3, dim: int = 768,
346                 depth: int = 12, heads: int = 12, dec_dim: int = 512, dec_depth: int = 8,
347                 dec_heads: int = 16, ratio: float = 0.75):
348        super().__init__()
349        N = (image_size // patch) ** 2
350        self.ratio = ratio
351        self.embed = PatchEmbed(patch, channels, dim)
352        self.positions = nn.Parameter(0.02 * torch.randn(1, N, dim))
353        self.encoder = nn.ModuleList(Block(dim, heads) for _ in range(depth))
354        self.norm = nn.LayerNorm(dim)
355        self.to_decoder = nn.Linear(dim, dec_dim)
356        self.mask_token = nn.Parameter(torch.zeros(1, 1, dec_dim))                 # m
357        self.dec_positions = nn.Parameter(0.02 * torch.randn(1, N, dec_dim))
358        self.decoder = nn.ModuleList(Block(dec_dim, dec_heads) for _ in range(dec_depth))
359        self.to_pixels = nn.Linear(dec_dim, patch * patch * channels)
360 
361    def forward(self, images: Tensor, generator: torch.Generator = None):
362        x = self.embed(images) + self.positions                        # positions before masking
363        h, mask, ids_restore = random_masking(x, self.ratio, generator)
364        for block in self.encoder:
365            h = block(h)                                               # [B, N/4, D] only
366        h = unshuffle(self.to_decoder(self.norm(h)), self.mask_token, ids_restore) + self.dec_positions
367        for block in self.decoder:
368            h = block(h)                                               # [B, N, D_dec]
369        pred = self.to_pixels(h)                                       # [B, N, P*P*C]
370        return mae_loss(pred, patchify(images, self.embed.patch), mask), pred, mask
371# END vit-mae

374# BEGIN vit-dino
375class DINOLoss(nn.Module):
376    """L = sum over teacher crops x and student crops x' != x of  -P_t(x)^T log P_s(x').
377 
378    P_s(x) = softmax(g_s(x) / tau_s);  P_t(x) = softmax((g_t(x) - c) / tau_t), centred and sharpened.
379    The centre c is an EMA of the teacher's batch mean. The sum is averaged over the pairs.
380    """
381 
382    def __init__(self, out_dim: int, tau_s: float = 0.1, tau_t: float = 0.04, center_momentum: float = 0.9):
383        super().__init__()
384        self.tau_s, self.tau_t, self.m = tau_s, tau_t, center_momentum
385        self.register_buffer("center", torch.zeros(1, out_dim))       # c
386 
387    def forward(self, student: list, teacher: list) -> Tensor:
388        """student: [B, K] outputs for every crop, global crops first; teacher: one per global crop."""
389        targets = [F.softmax((t - self.center) / self.tau_t, dim=-1).detach() for t in teacher]
390        total, pairs = 0.0, 0
391        for i, p_t in enumerate(targets):
392            for j, s in enumerate(student):
393                if j == i:                                             # the same view on both sides
394                    continue
395                total = total + (-p_t * F.log_softmax(s / self.tau_s, dim=-1)).sum(dim=-1).mean()
396                pairs += 1
397        self.update_center(torch.cat(teacher))
398        return total / pairs
399 
400    @torch.no_grad()
401    def update_center(self, teacher_out: Tensor) -> None:
402        """c <- m c + (1 - m) * mean of the teacher outputs in the batch."""
403        self.center.mul_(self.m).add_((1 - self.m) * teacher_out.mean(dim=0, keepdim=True))
404 
405 
406@torch.no_grad()
407def ema_update(teacher: nn.Module, student: nn.Module, lam: float) -> None:
408    """theta_t <- lam * theta_t + (1 - lam) * theta_s. The teacher never receives gradients."""
409    for t, s in zip(teacher.parameters(), student.parameters()):
410        t.mul_(lam).add_((1 - lam) * s.detach())

415class TubeletEmbed(nn.Module):
416    """ViViT: Conv3d with kernel = stride = (t, P, P), one token per t x P x P tube.
417    [B, C, T, H, W] -> [B, T/t, (H/P)*(W/P), D]."""
418 
419    def __init__(self, t: int, patch: int, channels: int, dim: int):
420        super().__init__()
421        self.proj = nn.Conv3d(channels, dim, kernel_size=(t, patch, patch), stride=(t, patch, patch))
422 
423    def forward(self, video: Tensor) -> Tensor:
424        x = self.proj(video)                                           # [B, D, T', H', W']
425        B, D, T, H, W = x.shape
426        return x.permute(0, 2, 3, 4, 1).reshape(B, T, H * W, D)

429class DividedSpaceTimeBlock(nn.Module):
430    """TimeSformer: attention over time (one patch position, all frames), then over space (one
431    frame, all patches), then the MLP, each pre-norm with a residual. x: [B, T, N, D]."""
432 
433    def __init__(self, dim: int, heads: int):
434        super().__init__()
435        self.norm_t, self.time = nn.LayerNorm(dim), Attention(dim, heads)
436        self.norm_s, self.space = nn.LayerNorm(dim), Attention(dim, heads)
437        self.norm_m, self.mlp = nn.LayerNorm(dim), mlp(dim)
438 
439    def mix_time(self, x: Tensor) -> Tensor:
440        B, T, N, D = x.shape
441        seqs = self.norm_t(x).transpose(1, 2).reshape(B * N, T, D)    # B*N sequences of length T
442        return x + self.time(seqs).view(B, N, T, D).transpose(1, 2)
443 
444    def mix_space(self, x: Tensor) -> Tensor:
445        B, T, N, D = x.shape
446        seqs = self.norm_s(x).reshape(B * T, N, D)                     # B*T sequences of length N
447        return x + self.space(seqs).view(B, T, N, D)
448 
449    def forward(self, x: Tensor) -> Tensor:
450        x = self.mix_space(self.mix_time(x))
451        return x + self.mlp(self.norm_m(x))

456def packing_mask(lengths: list) -> Tensor:
457    """NaViT: images of different sizes share one sequence of sum(lengths) tokens; the mask is
458    block-diagonal, so a token reads only the tokens of its own image. [L, L], True = may read."""
459    image = torch.repeat_interleave(torch.arange(len(lengths)), torch.tensor(lengths))
460    return image[:, None] == image[None, :]

463class FactorizedPositions(nn.Module):
464    """E_pos(y, x) = phi_y[y] + phi_x[x]: one table per axis, so any grid up to max_side on each
465    side and any aspect ratio has positions."""
466 
467    def __init__(self, max_side: int, dim: int):
468        super().__init__()
469        self.phi_y = nn.Parameter(0.02 * torch.randn(max_side, dim))
470        self.phi_x = nn.Parameter(0.02 * torch.randn(max_side, dim))
471 
472    def forward(self, rows: int, cols: int) -> Tensor:
473        return (self.phi_y[:rows, None] + self.phi_x[None, :cols]).reshape(rows * cols, -1)  # [rows*cols, D]

477# BEGIN vit-dit
478class DiTBlock(nn.Module):
479    """adaLN-Zero.  (beta1, gamma1, alpha1, beta2, gamma2, alpha2) = Linear(SiLU(c)), c = e_t + e_y;
480    x <- x + alpha1 * MSA((1 + gamma1) * LN(x) + beta1);  x <- x + alpha2 * MLP((1 + gamma2) * LN(x) + beta2).
481    The linear layer starts at zero, so every alpha is 0 and the block starts as the identity."""
482 
483    def __init__(self, dim: int, heads: int):
484        super().__init__()
485        self.norm1 = nn.LayerNorm(dim, elementwise_affine=False, eps=1e-6)
486        self.attn = Attention(dim, heads)
487        self.norm2 = nn.LayerNorm(dim, elementwise_affine=False, eps=1e-6)
488        self.mlp = mlp(dim)
489        self.ada = nn.Sequential(nn.SiLU(), nn.Linear(dim, 6 * dim))
490        nn.init.zeros_(self.ada[1].weight)
491        nn.init.zeros_(self.ada[1].bias)
492 
493    def forward(self, x: Tensor, c: Tensor) -> Tensor:                # x: [B, T, D], c: [B, D]
494        beta1, gamma1, alpha1, beta2, gamma2, alpha2 = self.ada(c)[:, None].chunk(6, dim=-1)
495        x = x + alpha1 * self.attn(self.norm1(x) * (1 + gamma1) + beta1)
496        return x + alpha2 * self.mlp(self.norm2(x) * (1 + gamma2) + beta2)
497# END vit-dit

501def attention_cost(N: int, D: int, M: int = None) -> tuple:
502    """Multiply-adds of one attention sublayer: 4 N D^2 for the projections, plus 2 N^2 D for the
503    scores and the weighted sum, or 2 M^2 N D when every query reads only its M x M window."""
504    return 4 * N * D * D, 2 * N * N * D if M is None else 2 * M * M * N * D
