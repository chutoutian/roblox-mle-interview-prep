1"""A minimal vision-language model that follows Chapter 3 equation by equation.
2 
3Layout (LLaVA): ViT vision encoder -> MLP projector -> decoder-only LLM.
4Run:   .venv-transformer/bin/python vlm_vit_projector_llm.py
5Tests: .venv-transformer/bin/python -m unittest -v test_vlm_vit_projector_llm
6 
7Attention is the module of Chapter 2 (transformer_encoder_decoder.py); masks are
8boolean and True means "this key may be read". One image per row, no padding.
9"""

22# BEGIN vlm-patches
23def patchify(images: Tensor, patch: int) -> Tensor:
24    """Step 1: [B, C, H, W] -> [B, N, P*P*C], patches in row-major order."""
25    B, C, H, W = images.shape                                     # PyTorch stores images channel-first
26    x = images.reshape(B, C, H // patch, patch, W // patch, patch)
27    x = x.permute(0, 2, 4, 3, 5, 1)                               # [B, H/P, W/P, P, P, C]
28    return x.reshape(B, (H // patch) * (W // patch), patch * patch * C)
29 
30 
31class PatchEmbedding(nn.Module):
32    """Step 2:  Z0 = X_p W_E + b_E + P_vis.  Equal to Conv2d(C, d_v, kernel=P, stride=P)."""
33 
34    def __init__(self, image_size: int, patch: int, channels: int, d_vision: int):
35        super().__init__()
36        self.patch = patch
37        self.linear = nn.Linear(patch * patch * channels, d_vision)                        # W_E, b_E
38        self.positions = nn.Parameter(0.02 * torch.randn((image_size // patch) ** 2, d_vision))  # P_vis
39 
40    def forward(self, images: Tensor) -> Tensor:
41        return self.linear(patchify(images, self.patch)) + self.positions                  # [B, N, d_v]
42# END vlm-patches

45# BEGIN vlm-blocks
46class Block(nn.Module):
47    """Steps 3 and 7, pre-norm:  x' = x + MHA(LN(x));  x = x' + FFN(LN(x'))."""
48 
49    def __init__(self, d_model: int, num_heads: int, d_ff: int):
50        super().__init__()
51        self.norm1 = nn.LayerNorm(d_model)
52        self.attention = MultiHeadAttention(d_model, num_heads)
53        self.norm2 = nn.LayerNorm(d_model)
54        self.ffn = nn.Sequential(nn.Linear(d_model, d_ff), nn.GELU(), nn.Linear(d_ff, d_model))
55 
56    def forward(self, x: Tensor, mask: Tensor = None) -> Tensor:
57        h = self.norm1(x)
58        x = x + self.attention(h, h, h, mask)   # mask=None: every token reads every token
59        return x + self.ffn(self.norm2(x))
60# END vlm-blocks

63# BEGIN vlm-vision
64class VisionEncoder(nn.Module):
65    """Steps 1-3: image -> one contextual vector per patch. No mask, so it is bidirectional."""
66 
67    def __init__(self, image_size: int = 8, patch: int = 4, channels: int = 3,
68                 d_vision: int = 8, num_heads: int = 2, num_layers: int = 2):
69        super().__init__()
70        self.d_vision = d_vision
71        self.embed = PatchEmbedding(image_size, patch, channels, d_vision)
72        self.layers = nn.ModuleList(Block(d_vision, num_heads, 4 * d_vision) for _ in range(num_layers))
73 
74    def forward(self, images: Tensor) -> Tensor:
75        z = self.embed(images)        # steps 1-2: [B, C, H, W] -> [B, N, d_v]
76        for layer in self.layers:
77            z = layer(z)              # step 3
78        return z                      # Z: [B, N, d_v]
79 
80 
81class Projector(nn.Module):
82    """Step 4:  U = GELU(Z W_1 + b_1) W_2 + b_2, from the vision width d_v to the LLM width d."""
83 
84    def __init__(self, d_vision: int, d_model: int):
85        super().__init__()
86        self.linear1 = nn.Linear(d_vision, d_model)
87        self.linear2 = nn.Linear(d_model, d_model)
88 
89    def forward(self, z: Tensor) -> Tensor:
90        return self.linear2(F.gelu(self.linear1(z)))   # [B, N, d_v] -> [B, N, d]
91# END vlm-vision
92

94# BEGIN vlm-model
95class VLM(nn.Module):
96    """Steps 1-8: image + text ids -> next-token logits over the vocabulary."""
97 
98    def __init__(self, vocab_size: int, image_token_id: int, d_model: int = 16, num_heads: int = 2,
99                 num_layers: int = 2, max_len: int = 64, vision: VisionEncoder = None):
100        super().__init__()
101        self.image_token_id = image_token_id
102        self.vision = vision if vision is not None else VisionEncoder()
103        self.projector = Projector(self.vision.d_vision, d_model)
104        self.embedding = nn.Embedding(vocab_size, d_model)                      # E
105        self.positions = nn.Parameter(0.02 * torch.randn(max_len, d_model))     # P_txt (real LLMs: RoPE)
106        self.layers = nn.ModuleList(Block(d_model, num_heads, 4 * d_model) for _ in range(num_layers))
107        self.norm = nn.LayerNorm(d_model)
108        self.vocab_projection = nn.Linear(d_model, vocab_size, bias=False)      # W_vocab
109 
110    def encode_image(self, images: Tensor) -> Tensor:
111        return self.projector(self.vision(images))                 # steps 1-4: [B, C, H, W] -> [B, N, d]
112 
113    def splice(self, text_ids: Tensor, image_tokens: Tensor) -> Tensor:
114        """Steps 5-6: embed the text, then put the N image tokens where <image> stands."""
115        text = self.embedding(text_ids)                                         # T: [B, m, d]
116        rows = []
117        for b in range(text_ids.shape[0]):
118            k = (text_ids[b] == self.image_token_id).nonzero()[0, 0]            # position of <image>
119            rows.append(torch.cat([text[b, :k], image_tokens[b], text[b, k + 1:]]))
120        x = torch.stack(rows)                                                   # [B, L, d], L = m - 1 + N
121        return x + self.positions[:x.shape[1]]
122 
123    def splice_labels(self, text_ids: Tensor, text_labels: Tensor, num_image_tokens: int) -> Tensor:
124        """Labels for the spliced sequence: image positions are never trained on."""
125        ignore = torch.full((num_image_tokens,), IGNORE, dtype=text_labels.dtype, device=text_labels.device)
126        rows = []
127        for b in range(text_ids.shape[0]):
128            k = (text_ids[b] == self.image_token_id).nonzero()[0, 0]
129            rows.append(torch.cat([text_labels[b, :k], ignore, text_labels[b, k + 1:]]))
130        return torch.stack(rows)                                                # [B, L]
131 
132    def forward(self, images: Tensor, text_ids: Tensor, image_tokens: Tensor = None) -> Tensor:
133        if image_tokens is None:
134            image_tokens = self.encode_image(images)
135        x = self.splice(text_ids, image_tokens)                                 # X0: [B, L, d]
136        L = x.shape[1]
137        causal = torch.ones(L, L, dtype=torch.bool, device=x.device).tril()[None, None]  # [1, 1, L, L]
138        for layer in self.layers:
139            x = layer(x, causal)                                                # step 7
140        return self.vocab_projection(self.norm(x))                              # step 8: [B, L, V]
141

143def answer_loss(logits: Tensor, labels: Tensor) -> Tensor:
144    """Step 9: cross-entropy averaged over the answer positions only."""
145    return F.cross_entropy(logits.reshape(-1, logits.shape[-1]), labels.reshape(-1), ignore_index=IGNORE)
146 
147 
148def set_stage(model: VLM, stage: int) -> None:
149    """Step 10: which parameters are updated. The vision encoder stays frozen throughout.
150 
151    0 = the LLM alone (stands in for language pretraining), 1 = the projector alone (alignment),
152    2 = projector and LLM (instruction tuning).
153    """
154    for parameter in model.parameters():
155        parameter.requires_grad = stage in (0, 2)
156    for parameter in model.vision.parameters():
157        parameter.requires_grad = False
158    for parameter in model.projector.parameters():
159        parameter.requires_grad = stage in (1, 2)
160 
161 
162@torch.no_grad()
163def generate(model: VLM, images: Tensor, prompt_ids: Tensor, eos_id: int = 2, pad_id: int = 0,
164             max_new_tokens: int = 8) -> Tensor:
165    """Step 11: encode the image once, then append argmax tokens until every row produced EOS."""
166    image_tokens = model.encode_image(images)
167    ids = prompt_ids
168    finished = torch.zeros(ids.shape[0], dtype=torch.bool, device=ids.device)
169    for _ in range(max_new_tokens):
170        next_id = model(images, ids, image_tokens)[:, -1].argmax(dim=-1)        # last row of Z only
171        next_id = torch.where(finished, pad_id, next_id)
172        ids = torch.cat([ids, next_id[:, None]], dim=1)
173        finished |= next_id == eos_id
174        if finished.all():
175            break
176    return ids

181VOCAB = ["<pad>", "<bos>", "<eos>", "<image>", "what", "is", "this", "?", "a", "cat", "dog", "photo", "of"]

184def toy_batch():
185    """Two 8x8 images that differ in which patch is bright, the same question, different answers."""
186    cat, dog = torch.zeros(3, 8, 8), torch.zeros(3, 8, 8)
187    cat[:, :4, :4] = 1.0                                   # top-left patch bright     -> "a cat"
188    dog[:, 4:, 4:] = 1.0                                   # bottom-right patch bright -> "a dog"
189    full = torch.tensor([[1, 3, 4, 5, 6, 7, 8, 9, 2],      # BOS <image> what is this ? a cat EOS
190                         [1, 3, 4, 5, 6, 7, 8, 10, 2]])    # BOS <image> what is this ? a dog EOS
191    text_ids, next_ids = full[:, :-1], full[:, 1:].clone() # input and next token at every position
192    next_ids[:, :-3] = IGNORE                              # only the answer (a, cat/dog, EOS) is a label
193    captions = torch.tensor([[11, 12, 8, 9],               # photo of a cat
194                             [11, 12, 8, 10]])             # photo of a dog
195    return torch.stack([cat, dog]), text_ids, next_ids, captions

198def words(ids) -> str:
199    return " ".join(VOCAB[i] for i in ids)

202def train(model: VLM, stage: int, forward, labels: Tensor, steps: int, lr: float = 1e-2):
203    """Minimize the answer loss over the parameters that the stage leaves trainable."""
204    set_stage(model, stage)
205    trainable = [parameter for parameter in model.parameters() if parameter.requires_grad]
206    optimizer = torch.optim.Adam(trainable, lr=lr)
207    before = answer_loss(forward(), labels).item()
208    for _ in range(steps):
209        optimizer.zero_grad()
210        loss = answer_loss(forward(), labels)
211        loss.backward()                                    # gradients pass through the frozen parts
212        optimizer.step()
213    return sum(parameter.numel() for parameter in trainable), before, loss.item()

216def main() -> None:
217    torch.manual_seed(0)
218    images, text_ids, text_labels, captions = toy_batch()
219    model = VLM(vocab_size=len(VOCAB), image_token_id=3)
220    I, t = images[:1], text_ids[:1]                        # the cat example, B = 1
221    print("t =", t[0].tolist(), "=", words(t[0]), "\n")
222 
223    def show(step: str, name: str, tensor: Tensor) -> Tensor:
224        print(f"step {step:<3} {name:<34} {list(tensor.shape)}")
225        return tensor
226 
227    show("0", "I (channel-first)", I)
228    show("1", "X_p = patchify(I)", patchify(I, 4))
229    show("2", "Z0 = X_p W_E + b_E + P_vis", model.vision.embed(I))
230    Z = show("3", "Z = ViT(Z0)", model.vision(I))
231    show("3", "    attention weights, per head", model.vision.layers[-1].attention.attention_weights)
232    U = show("4", "U = projector(Z)", model.projector(Z))
233    show("5", "T = E[t]", model.embedding(t))
234    show("6", "X0 = splice(T, U) + P_txt", model.splice(t, U))
235    logits = model(I, t)
236    show("7", "    causal weights, per head", model.layers[-1].attention.attention_weights)
237    show("8", "Z_out = LN(X) W_vocab", logits)
238 
239    labels = model.splice_labels(text_ids, text_labels, U.shape[1])             # both examples: [2, 11]
240    print("\nlabels y =", labels[0].tolist())
241    print(f"step 9   loss over positions 8, 9, 10 = {answer_loss(logits, labels[:1]).item():.4f}  (untrained)\n")
242 
243    total = sum(parameter.numel() for parameter in model.parameters())
244    with_words = lambda: model(None, text_ids, model.embedding(captions))       # a caption fills the image slot
245    with_images = lambda: model(images, text_ids)                               # projected patches fill it
246    for stage, forward, steps, note in ((0, with_words, 300, "LLM, caption words in the image slot"),
247                                        (1, with_images, 300, "projector only, real image tokens"),
248                                        (2, with_images, 100, "projector + LLM")):
249        count, before, after = train(model, stage, forward, labels, steps)
250        print(f"step 10  stage {stage} ({note}): {count} of {total} parameters, loss {before:.3f} -> {after:.3f}")
251 
252    model.eval()
253    prompt = text_ids[:, :6]                               # BOS <image> what is this ?
254    for name, ids in zip(("cat image", "dog image"), generate(model, images, prompt)):
255        print(f"step 11  {name}: {words(ids.tolist())}")

259    main()
