4class IntraPredictor:
5    """
6    Interview-style intra prediction demo for one NxN block.
7    Modes:
8      1) DC
9      2) Planar (HEVC-like simplified integer form)
10      3) Angular (simplified: vertical-like / horizontal-like with interpolation)
11 
12    Assumptions:
13    - top and left have length N
14    - pixel values are integers (typically 0..255)
15    - for planar, tr and bl are provided (top-right / bottom-left extension)
16    """
17 
18    @staticmethod
19    def clip_pixel(v: int, lo: int = 0, hi: int = 255) -> int:
20        return max(lo, min(hi, v))
21 
22    @staticmethod
23    def dc_predict(top: List[int], left: List[int]) -> List[List[int]]:
24        n = len(top)
25        if len(left) != n:
26            raise ValueError("top and left must have same length")
27 
28        # DC = round((sum(top)+sum(left)) / (2n)) with integer-style rounding
29        dc = (sum(top) + sum(left) + n) // (2 * n)
30        dc = IntraPredictor.clip_pixel(dc)
31 
32        return [[dc for _ in range(n)] for _ in range(n)]
33 
34    @staticmethod
35    def planar_predict(
36        top: List[int],
37        left: List[int],
38        tr: int,
39        bl: int
40    ) -> List[List[int]]:
41        """
42        HEVC-style planar intuition:
43          P(x,y) = ((N-1-x)*L[y] + (x+1)*TR + (N-1-y)*T[x] + (y+1)*BL + N) // (2N)
44        """
45        n = len(top)
46        if len(left) != n:
47            raise ValueError("top and left must have same length")
48 
49        out = [[0] * n for _ in range(n)]
50        denom = 2 * n
51 
52        for y in range(n):
53            for x in range(n):
54                num = (
55                    (n - 1 - x) * left[y]
56                    + (x + 1) * tr
57                    + (n - 1 - y) * top[x]
58                    + (y + 1) * bl
59                    + n
60                )
61                out[y][x] = IntraPredictor.clip_pixel(num // denom)
62        return out
63 
64    @staticmethod
65    def _interp_1d(ref: List[int], pos_num: int, scale: int = 32) -> int:
66        """
67        Linear interpolation on ref at fixed-point position pos_num/scale.
68        Example:
69          pos_num = i*scale + w, w in [0, scale-1]
70          val = ((scale-w)*ref[i] + w*ref[i+1] + scale//2) // scale
71        with boundary clamping.
72        """
73        i = pos_num // scale
74        w = pos_num % scale
75 
76        i0 = max(0, min(i, len(ref) - 1))
77        i1 = max(0, min(i + 1, len(ref) - 1))
78 
79        val = ((scale - w) * ref[i0] + w * ref[i1] + scale // 2) // scale
80        return IntraPredictor.clip_pixel(val)
81 
82    @staticmethod
83    def angular_predict(
84        top: List[int],
85        left: List[int],
86        mode: str = "vertical",
87        d: int = 0
88    ) -> List[List[int]]:
89        """
90        Simplified angular predictor with fixed-point interpolation.
91 
92        mode="vertical":
93            project from top refs, with slope controlled by d
94            pos = x + (y+1)*d/32
95        mode="horizontal":
96            project from left refs, with slope controlled by d
97            pos = y + (x+1)*d/32
98 
99        d > 0: tilt one way, d < 0: tilt the other way, d=0: pure vertical/horizontal copy.
100        """
101        n = len(top)
102        if len(left) != n:
103            raise ValueError("top and left must have same length")
104        if mode not in ("vertical", "horizontal"):
105            raise ValueError("mode must be 'vertical' or 'horizontal'")
106 
107        out = [[0] * n for _ in range(n)]
108        scale = 32
109 
110        if mode == "vertical":
111            # Reference line for x direction
112            ref = top[:]  # length n
113            for y in range(n):
114                for x in range(n):
115                    # fixed-point position numerator: (x * 32 + (y+1)*d)
116                    pos_num = x * scale + (y + 1) * d
117                    out[y][x] = IntraPredictor._interp_1d(ref, pos_num, scale)
118        else:
119            # horizontal: use left refs
120            ref = left[:]
121            for y in range(n):
122                for x in range(n):
123                    # fixed-point position numerator: (y * 32 + (x+1)*d)
124                    pos_num = y * scale + (x + 1) * d
125                    out[y][x] = IntraPredictor._interp_1d(ref, pos_num, scale)
126 
127        return out
128 
129    @staticmethod
130    def residual(block: List[List[int]], pred: List[List[int]]) -> List[List[int]]:
131        n = len(block)
132        if n == 0 or any(len(row) != n for row in block):
133            raise ValueError("block must be NxN")
134        if len(pred) != n or any(len(row) != n for row in pred):
135            raise ValueError("pred must be NxN and same size as block")
136 
137        return [[block[y][x] - pred[y][x] for x in range(n)] for y in range(n)]
138 
139    @staticmethod
140    def sad(block: List[List[int]], pred: List[List[int]]) -> int:
141        r = IntraPredictor.residual(block, pred)
142        return sum(abs(v) for row in r for v in row)
143 
144    @staticmethod
145    def choose_mode_by_sad(
146        block: List[List[int]],
147        top: List[int],
148        left: List[int],
149        tr: int,
150        bl: int
151    ) -> Tuple[str, List[List[int]], int]:
152        """
153        Interview-friendly mode decision using SAD only (not full RD).
154        Returns: (best_mode_name, best_prediction, best_sad)
155        """
156        candidates = {}
157 
158        p_dc = IntraPredictor.dc_predict(top, left)
159        candidates["DC"] = (p_dc, IntraPredictor.sad(block, p_dc))
160 
161        p_planar = IntraPredictor.planar_predict(top, left, tr, bl)
162        candidates["Planar"] = (p_planar, IntraPredictor.sad(block, p_planar))
163 
164        # Try a few angular candidates (you can add more)
165        p_ang_v0 = IntraPredictor.angular_predict(top, left, mode="vertical", d=0)
166        candidates["Angular-Vertical(d=0)"] = (p_ang_v0, IntraPredictor.sad(block, p_ang_v0))
167 
168        p_ang_v16 = IntraPredictor.angular_predict(top, left, mode="vertical", d=16)
169        candidates["Angular-Vertical(d=16)"] = (p_ang_v16, IntraPredictor.sad(block, p_ang_v16))
170 
171        p_ang_h16 = IntraPredictor.angular_predict(top, left, mode="horizontal", d=16)
172        candidates["Angular-Horizontal(d=16)"] = (p_ang_h16, IntraPredictor.sad(block, p_ang_h16))
173 
174        best_mode, (best_pred, best_cost) = min(candidates.items(), key=lambda kv: kv[1][1])
175        return best_mode, best_pred, best_cost
