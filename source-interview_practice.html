15def psnr(raw_image, processed_image):
16    
17 
18    # psnr = 10 * log10(MAX^2 / MSE)
19 
20    mse = np.mean( (raw_image - processed_image) ** 2 )
21    max_pixel = 255.0
22 
23    if mse == 0:
24        return float('inf')
25    else:
26        psnr = 10 * np.log10(max_pixel ** 2 / mse)

42def convlution(image, kernel, stride = 1):
43    h, w = image.shape
44    kh, kw = kernel.shape
45    
46    ## version 1: no padding
47    out_h = (h - kh) // stride + 1
48    out_w = (w - kw) // stride + 1
49 
50    output = np.zeros( (out_h, out_w), dtype = np.float32)
51    for i in range(out_h):
52        for j in range(out_w):
53 
54            output[i,j] = np.sum(image[i*stride:i*stride+kh, j*stride:j*stride+kw] * kernel)
55    
56    
57    ### version 2: with padding
58 
59    pad_h = kh // 2 #
60    pad_w = kw // 2 #
61 
62    padded_image = np.zeros( (h + 2*pad_h, w+ 2*pad_w), dtype = np.float32)
63    
64    padded_image[pad_h:pad_h+h, pad_w:pad_w+w] = image
65    # padding should be the value of the border pixel, not zero
66    padded_image[:pad_h, pad_w:pad_w+w] = image[0:pad_h, :]
67    padded_image[pad_h+h:, pad_w:pad_w+w] = image[h-pad_h:, :]
68    padded_image[:, :pad_w] = padded_image[:, pad_w:pad_w+1]
69    padded_image[:, pad_w+w:] = padded_image[:, pad_w+w-1:pad_w+w]
70 
71    out_h = (h + 2*pad_h - kh) // stride + 1
72    out_w = (w + 2*pad_w - kw) // stride + 1
73 
74    output = np.zeros( (out_h, out_w), dtype = np.float32)
75    for i in range(out_h):
76        for j in range(out_w):
77 
78            output[i,j] = np.sum(padded_image[i*stride:i*stride+kh, j*stride:j*stride+kw] * kernel)
79 
80    return output

82def convolve(
83    image:   List[List[float]],
84    kernel:  List[List[float]],
85    stride:  int = 1,
86    padding: str = "same",    # "valid" | "same"
87) -> List[List[float]]:
88    H,  W  = len(image),  len(image[0])
89    kH, kW = len(kernel), len(kernel[0])
90 
91    # --- padding ---
92    if padding == "same":
93        pad_h, pad_w = (kH - 1) // 2, (kW - 1) // 2
94        padded = [[0.0] * (W + 2 * pad_w) for _ in range(H + 2 * pad_h)]
95        for i in range(H):
96            for j in range(W):
97                padded[i + pad_h][j + pad_w] = image[i][j]
98    else:                          # "valid": no padding
99        pad_h, pad_w = 0, 0
100        padded = image
101 
102    pH, pW = len(padded), len(padded[0])
103 
104    # --- output size ---
105    out_H = (pH - kH) // stride + 1
106    out_W = (pW - kW) // stride + 1
107 
108    result = [[0.0] * out_W for _ in range(out_H)]
109 
110    # --- slide kernel over image ---
111    for i in range(out_H):
112        for j in range(out_W):
113            total = 0.0
114            for ki in range(kH):
115                for kj in range(kW):
116                    total += padded[i * stride + ki][j * stride + kj] * kernel[ki][kj]
117            result[i][j] = total
118 
119    return result

138def rotation(image, degree):
139    
140    h, w = image.shape
141 
142    # step 1: transpose
143    for i in range(h):
144        for j in range(i+1, w):
145            image[i,j], image[j,i] = image[j,i], image[i,j]
146    
147    # step 2: reverse each row
148    for i in range(h):
149        image[i] = image[i][::-1]
150    
151    return image

154def rotation_any_degree(image, degree):
155    # this is more complicated, we can use inverse mapping to find the source pixel for each output pixel
156    h, w = image.shape
157    rad = np.radians(degree)
158    cos_a, sin_a = np.cos(rad), np.sin(rad)
159 
160    new_h = int(abs(h * cos_a) + abs(w * sin_a))
161    new_w = int(abs(h * sin_a) + abs(w * cos_a))
162 
163    cx_in, cy_in = (w - 1) / 2.0, (h - 1) / 2.0
164    cx_out, cy_out = (new_w - 1) / 2.0, (new_h - 1) / 2.0
165 
166    res = np.zeros( (new_h, new_w), dtype = np.float32)
167 
168    for i in range(new_h):
169        for j in range(new_w):
170            x = j - cx_out
171            y = i - cy_out
172 
173            src_x = int(x * cos_a + y * sin_a + cx_in)
174            src_y = int(-x * sin_a + y * cos_a + cy_in)
175 
176            if 0 <= src_x < w and 0 <= src_y < h:
177                res[i,j] = image[src_y, src_x]
178    return result

184def motion_vector(reference, image, block_size):
185    # this is a very complex problem, we can use block matching algorithm to find the motion vector for each block
186    w, h = image.shape
187    res = np.zeros( (w//block_size, h//block_size, 2), dtype = np.float32)
188    for i in range(0,w,block_size):
189        for j in range(0, h ,block_size):
190            best_asd = float('inf')
191            best_mv = (0,0)
192            for mv_x in range(-block_size, block_size+1):
193                for mv_y in range(-block_size, block_size+1):
194                    asd = 0.0
195                    for k in range(block_size):
196                        for l in range(block_size):
197                            if 0<= i+mv_x < w and 0 <= j+mv_y <h:
198                                reference_block = reference[i+mv_x+k, j+mv_y+l]
199                                current_block = image[i+k, j+l]
200                                asd += abs(reference_block - current_block)
201                    if asd < best_asd:
202                        best_asd = asd
203                        best_mv = (mv_x, mv_y)  
204            res[i//block_size, j//block_size] = best_mv
205    return res

212def huffman_encoding(symbols):
213    
214    symbol_count = Counter(symbols)
215    heap = []
216    order = 0
217    for sym, c in symbol_count.items():
218        heappush(heap, (int(c), order, int(sym)))
219        order += 1
220    
221    while len(heap) > 1:
222        f1, _, s1 = heappop(heap)
223        f2, _, s2 = heappop(heap)
224        heappush(heap, (f1+ f2, order, (s1,s2)))
225        order += 1
226    codebook = {}
227    def dfs(node, path):
228        if isinstance(node, int):
229            codebook[node] = path
230            return
231        left, right = node
232        dfs(left, path + "0")
233        dfs(right, path + "1")
234        return codebook
235    root = heap[0][2] if heap else None
236    dfs(root, "")

249def zig_zag_encode(arr):
250    m = len(arr)
251    n = len(arr[0])
252    lay = 0
253    result = []
254    for lay in range(m + n - 1):
255        if lay % 2 == 0:
256            # Even layer: traverse from bottom-left to top-right
257            r = min(lay, m - 1)
258            c = lay - r
259            while r >= 0 and c < n:
260                result.append(arr[r][c])
261                r -= 1
262                c += 1
263        else:
264            # Odd layer: traverse from top-right to bottom-left
265            c = min(lay, n - 1)
266            r = lay - c
267            while c >= 0 and r < m:
268                result.append(arr[r][c])
269                r += 1
270                c -= 1
271    return result
