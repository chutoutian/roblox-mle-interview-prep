20class CausalSelfAttention(nn.Module):
21    """Step 2 of Chapter 2 with a causal mask, written so that past keys and values are reused."""
22 
23    def __init__(self, d_model: int, num_heads: int):
24        super().__init__()
25        self.num_heads, self.d_k = num_heads, d_model // num_heads
26        self.w_q = nn.Linear(d_model, d_model, bias=False)
27        self.w_k = nn.Linear(d_model, d_model, bias=False)
28        self.w_v = nn.Linear(d_model, d_model, bias=False)
29        self.w_o = nn.Linear(d_model, d_model, bias=False)
30 
31    def forward(self, x: Tensor, cache: dict | None = None) -> tuple[Tensor, dict]:
32        batch, t_new, d_model = x.shape
33        heads = lambda z: z.view(batch, t_new, self.num_heads, self.d_k).transpose(1, 2)  # [B, h, t_new, d_k]
34        q, k, v = heads(self.w_q(x)), heads(self.w_k(x)), heads(self.w_v(x))
35        if cache is not None:                        # step 3: the past is read, never recomputed
36            k = torch.cat([cache["k"], k], dim=2)
37            v = torch.cat([cache["v"], v], dim=2)
38        t_past = k.shape[2] - t_new
39        scores = q @ k.transpose(-2, -1) / math.sqrt(self.d_k)      # [B, h, t_new, t_past + t_new]
40        causal = torch.ones(t_new, t_past + t_new, dtype=torch.bool, device=x.device).tril(diagonal=t_past)
41        scores = scores.masked_fill(~causal, float("-inf"))
42        out = (scores.softmax(dim=-1) @ v).transpose(1, 2).reshape(batch, t_new, d_model)
43        return self.w_o(out), {"k": k, "v": v}

46class Block(nn.Module):
47    """Pre-norm block:  x' = x + Attention(LN(x));  x'' = x' + FFN(LN(x'))."""
48 
49    def __init__(self, d_model: int, num_heads: int, d_ff: int):
50        super().__init__()
51        self.norm1 = nn.LayerNorm(d_model)
52        self.attention = CausalSelfAttention(d_model, num_heads)
53        self.norm2 = nn.LayerNorm(d_model)
54        self.ffn = nn.Sequential(nn.Linear(d_model, d_ff), nn.GELU(), nn.Linear(d_ff, d_model))
55 
56    def forward(self, x: Tensor, cache: dict | None = None) -> tuple[Tensor, dict]:
57        a, cache = self.attention(self.norm1(x), cache)
58        x = x + a
59        return x + self.ffn(self.norm2(x)), cache

62def sinusoidal_positions(max_len: int, d_model: int) -> Tensor:
63    """The fixed table P of Chapter 2, step 1."""
64    position = torch.arange(max_len, dtype=torch.float64)[:, None]
65    frequency = 10000.0 ** (-torch.arange(0, d_model, 2, dtype=torch.float64) / d_model)
66    table = torch.zeros(max_len, d_model, dtype=torch.float64)
67    table[:, 0::2], table[:, 1::2] = torch.sin(position * frequency), torch.cos(position * frequency)
68    return table

71class CausalLM(nn.Module):
72    """A decoder-only Transformer: embeddings + positions, L pre-norm blocks, final norm, LM head."""
73 
74    def __init__(self, vocab_size: int, d_model: int = 32, num_heads: int = 4, d_ff: int = 64,
75                 num_layers: int = 2, max_len: int = 256):
76        super().__init__()
77        self.d_model = d_model
78        self.embedding = nn.Embedding(vocab_size, d_model)
79        self.register_buffer("positions", sinusoidal_positions(max_len, d_model).float())
80        self.blocks = nn.ModuleList(Block(d_model, num_heads, d_ff) for _ in range(num_layers))
81        self.norm = nn.LayerNorm(d_model)
82        self.lm_head = nn.Linear(d_model, vocab_size, bias=False)
83 
84    def forward(self, ids: Tensor, caches: list | None = None, start: int = 0) -> tuple[Tensor, list]:
85        """ids: [B, t_new] token ids that sit at positions start .. start + t_new - 1.
86        Returns the logits of every new position, [B, t_new, V], and the updated cache of every layer."""
87        x = self.embedding(ids) * math.sqrt(self.d_model) + self.positions[start:start + ids.shape[1]].to(ids.device)
88        updated = []
89        for i, block in enumerate(self.blocks):
90            x, cache = block(x, None if caches is None else caches[i])
91            updated.append(cache)
92        return self.lm_head(self.norm(x)), updated

96# BEGIN inference-cache
97def prefill(model: CausalLM, ids: Tensor) -> tuple[Tensor, list]:
98    """Step 1: the whole prompt in one pass. Returns the logits of the last position and every layer's cache."""
99    logits, caches = model(ids)
100    return logits[:, -1], caches
101

103def decode_step(model: CausalLM, token: Tensor, caches: list) -> tuple[Tensor, list]:
104    """Step 3: one new token reads the cache and appends its own key and value. token: [B]."""
105    start = caches[0]["k"].shape[2]
106    logits, caches = model(token[:, None], caches, start=start)
107    return logits[:, -1], caches
108

110def cache_bytes(num_layers: int, kv_heads: int, d_head: int, tokens: int, batch: int = 1, bytes_per_value: int = 2) -> int:
111    """Step 2: 2 (K and V) x layers x KV heads x d_head x tokens x batch x bytes per value."""
112    return 2 * num_layers * kv_heads * d_head * tokens * batch * bytes_per_value
113

115@torch.no_grad()
116def generate(model: CausalLM, prompt: Tensor, max_new_tokens: int, sample, use_cache: bool = True) -> Tensor:
117    """Steps 1 and 3 in a loop: prefill once, then one decode step per token. `sample` maps logits [B, V] to tokens [B]."""
118    ids = prompt
119    logits, caches = prefill(model, ids)
120    for _ in range(max_new_tokens):
121        token = sample(logits)
122        ids = torch.cat([ids, token[:, None]], dim=1)
123        if use_cache:
124            logits, caches = decode_step(model, token, caches)
125        else:                                        # the quadratic alternative: rerun the whole prefix
126            logits = model(ids)[0][:, -1]
127    return ids
128# END inference-cache

131# BEGIN inference-sampling
132def filtered_probabilities(logits: Tensor, temperature: float = 1.0, top_k: int | None = None,
133                           top_p: float | None = None, min_p: float | None = None) -> Tensor:
134    """Step 5: temperature, then truncation, then renormalisation. logits [B, V] -> probabilities [B, V]."""
135    if temperature <= 0:                             # the T -> 0 limit: all mass on the arg max (greedy)
136        return nn.functional.one_hot(logits.argmax(dim=-1), logits.shape[-1]).to(logits.dtype)
137    probs = (logits / temperature).softmax(dim=-1)
138    keep = torch.ones_like(probs, dtype=torch.bool)
139    if top_k is not None:                            # the k largest probabilities
140        keep &= probs >= probs.topk(top_k, dim=-1).values[..., -1:]
141    if top_p is not None:                            # the smallest prefix of the sorted list whose mass reaches p
142        sorted_probs, order = probs.sort(dim=-1, descending=True)
143        mass_before = sorted_probs.cumsum(dim=-1) - sorted_probs
144        keep &= torch.zeros_like(keep).scatter(-1, order, mass_before < top_p)
145    if min_p is not None:                            # at least a fraction min_p of the largest probability
146        keep &= probs >= min_p * probs.amax(dim=-1, keepdim=True)
147    probs = probs * keep
148    return probs / probs.sum(dim=-1, keepdim=True)

151def repetition_penalty(logits: Tensor, context: Tensor, penalty: float) -> Tensor:
152    """Logits of tokens already in the context are divided by `penalty` when positive and multiplied when negative."""
153    logits = logits.clone()
154    for b in range(logits.shape[0]):
155        seen = context[b].unique()
156        z = logits[b, seen]
157        logits[b, seen] = torch.where(z > 0, z / penalty, z * penalty)
158    return logits
159

161def sample_next(logits: Tensor, temperature: float = 1.0, top_k: int | None = None, top_p: float | None = None,
162                min_p: float | None = None, generator: torch.Generator | None = None) -> Tensor:
163    probs = filtered_probabilities(logits, temperature, top_k, top_p, min_p)
164    return torch.multinomial(probs, 1, generator=generator).squeeze(-1)
165# END inference-sampling

169# BEGIN inference-demo
170CORPUS = ("the cat sat on the mat . the dog sat on the log . "
171          "the cat saw the dog on the log . the dog saw the cat on the mat .")
172WORDS = sorted(set(CORPUS.split()))
173IDS = torch.tensor([[WORDS.index(w) for w in CORPUS.split()]])
174 
175 
176def train(model: CausalLM, steps: int, lr: float = 5e-3) -> float:
177    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
178    for _ in range(steps):
179        logits, _ = model(IDS[:, :-1])
180        loss = nn.functional.cross_entropy(logits.reshape(-1, logits.shape[-1]), IDS[:, 1:].reshape(-1))
181        optimizer.zero_grad(), loss.backward(), optimizer.step()
182    return loss.item()

186    return " ".join(WORDS[i] for i in ids[0].tolist())
