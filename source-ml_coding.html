31def binary_cross_entropy(y_pred: np.ndarray, y_true: np.ndarray,
32                         eps: float = 1e-9) -> float:
33    """
34    Binary cross-entropy for sigmoid output.
35    y_pred: probabilities in (0, 1)
36    y_true: binary labels {0, 1}
37    """
38    y_pred = np.clip(y_pred, eps, 1 - eps)
39    return -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))

42def cross_entropy_loss(logits: np.ndarray, labels: np.ndarray) -> float:
43    """
44    Categorical cross-entropy from raw logits (applies softmax internally).
45    logits: shape [N, C]
46    labels: integer class indices, shape [N]
47    """
48    # numerically stable softmax
49    shifted = logits - logits.max(axis=1, keepdims=True)
50    print(f"shifted logits:\n{shifted}")
51    log_sum_exp = np.log(np.sum(np.exp(shifted), axis=1, keepdims=True))
52    print(f"log_sum_exp:\n{log_sum_exp}")
53    log_probs = shifted - log_sum_exp                           # [N, C]
54    print(f"log_probs:\n{log_probs}")
55    return -np.mean(log_probs[np.arange(len(labels)), labels])

59def cross_entropy_loss(logits: np.ndarray, labels: np.ndarray) -> float:
60    log_sum_exp = np.log(np.sum(np.exp(logits), axis=1, keepdims=True))
61 
62    log_probs = logits - log_sum_exp                           # [N, C]
63    cross_entropy = 0
64    for i in range(len(labels)):
65        cross_entropy += -log_probs[i, labels[i]]
66    return cross_entropy / len(labels)

69def rate_distortion_loss(distortion: float, rate: float, lam: float) -> float:
70    """
71    Standard learned compression objective.
72    distortion: e.g. MSE between input and reconstruction
73    rate:       estimated bits (entropy of latent)
74    lam:        lagrange multiplier controlling bitrate-quality tradeoff
75                larger lam → penalise bits more → lower bitrate
76    """
77    return distortion + lam * rate

101def softmax(x: np.ndarray) -> np.ndarray:
102    """
103    Numerically stable softmax.
104    x: shape [N, C] or [C]
105    """
106    shifted = x - x.max(axis=-1, keepdims=True)
107    exp_x = np.exp(shifted)
108    return exp_x / exp_x.sum(axis=-1, keepdims=True)

111def gelu(x: np.ndarray) -> np.ndarray:
112    """
113    Gaussian Error Linear Unit — common in transformers.
114    Approximation: 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
115    """
116    return 0.5 * x * (1 + np.tanh(np.sqrt(2 / np.pi) * (x + 0.044715 * x ** 3)))

123class SGD:
124    def __init__(self, lr: float = 0.01, momentum: float = 0.0):
125        self.lr = lr
126        self.momentum = momentum
127        self.velocity = {}
128 
129    def step(self, params: dict, grads: dict):
130        for name in params:
131            if name not in self.velocity:
132                self.velocity[name] = np.zeros_like(params[name])
133            self.velocity[name] = (self.momentum * self.velocity[name]
134                                   - self.lr * grads[name])
135            params[name] += self.velocity[name]

138class Adam:
139    """
140    Adam optimiser: adaptive per-parameter learning rates.
141    m = first moment (mean of gradients)
142    v = second moment (mean of squared gradients)
143    bias-corrected estimates used for update.
144    """
145    def __init__(self, lr: float = 1e-3, beta1: float = 0.9,
146                 beta2: float = 0.999, eps: float = 1e-8):
147        self.lr = lr
148        self.beta1 = beta1
149        self.beta2 = beta2
150        self.eps = eps
151        self.m = {}
152        self.v = {}
153        self.t = 0
154 
155    def step(self, params: dict, grads: dict):
156        self.t += 1
157        for name in params:
158            if name not in self.m:
159                self.m[name] = np.zeros_like(params[name])
160                self.v[name] = np.zeros_like(params[name])
161 
162            self.m[name] = self.beta1 * self.m[name] + (1 - self.beta1) * grads[name]
163            self.v[name] = self.beta2 * self.v[name] + (1 - self.beta2) * grads[name] ** 2
164 
165            # bias correction
166            m_hat = self.m[name] / (1 - self.beta1 ** self.t)
167            v_hat = self.v[name] / (1 - self.beta2 ** self.t)
168 
169            params[name] -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

176class BatchNorm1d:
177    """
178    BatchNorm for shape [N, C].
179    Normalises each feature (column) across the batch.
180    Maintains running stats for inference.
181    """
182    def __init__(self, num_features: int, eps: float = 1e-5,
183                 momentum: float = 0.1):
184        self.eps = eps
185        self.momentum = momentum
186        self.gamma = np.ones(num_features)
187        self.beta = np.zeros(num_features)
188        self.running_mean = np.zeros(num_features)
189        self.running_var = np.ones(num_features)
190        # cache for backward
191        self._cache = None
192 
193    def forward(self, x: np.ndarray, training: bool = True) -> np.ndarray:
194        """x: [N, C]"""
195        if training:
196            mean = x.mean(axis=0)           # [C]
197            var = x.var(axis=0)             # [C]
198            x_hat = (x - mean) / np.sqrt(var + self.eps)
199            self._cache = (x, x_hat, mean, var)
200            # update running stats
201            self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * mean
202            self.running_var  = (1 - self.momentum) * self.running_var  + self.momentum * var
203        else:
204            x_hat = (x - self.running_mean) / np.sqrt(self.running_var + self.eps)
205        return self.gamma * x_hat + self.beta

208class LayerNorm:
209    """
210    LayerNorm: normalises each sample across its feature dimension.
211    No batch dependency — same behaviour at train and inference.
212    """
213    def __init__(self, normalized_shape: int, eps: float = 1e-5):
214        self.eps = eps
215        self.gamma = np.ones(normalized_shape)
216        self.beta = np.zeros(normalized_shape)
217 
218    def forward(self, x: np.ndarray) -> np.ndarray:
219        """x: [..., C] — normalises over last dimension."""
220        mean = x.mean(axis=-1, keepdims=True)
221        var = x.var(axis=-1, keepdims=True)
222        x_hat = (x - mean) / np.sqrt(var + self.eps)
223        return self.gamma * x_hat + self.beta

226def scaled_dot_product_attention(Q: np.ndarray, K: np.ndarray,
227                                  V: np.ndarray,
228                                  mask: np.ndarray = None) -> np.ndarray:
229    """
230    Attention(Q, K, V) = softmax(Q K^T / sqrt(d_k)) V
231    Q, K, V: shape [batch, seq_len, d_k] (or [seq_len, d_k] for single head)
232    mask:    additive mask (e.g. -inf for padding positions)
233    """
234    d_k = Q.shape[-1]
235    scores = Q @ K.swapaxes(-2, -1) / np.sqrt(d_k)    # [batch, seq, seq]
236    if mask is not None:
237        scores += mask
238    weights = softmax(scores)                           # [batch, seq, seq]
239    return weights @ V                                  # [batch, seq, d_v]

242class Linear:
243    """Fully connected layer with optional bias."""
244    def __init__(self, in_features: int, out_features: int, bias: bool = True):
245        # He initialisation
246        self.W = np.random.randn(in_features, out_features) * np.sqrt(2 / in_features)
247        self.b = np.zeros(out_features) if bias else None
248        self._cache = None
249 
250    def forward(self, x: np.ndarray) -> np.ndarray:
251        self._cache = x
252        out = x @ self.W
253        if self.b is not None:
254            out += self.b
255        return out
256 
257    def backward(self, grad_out: np.ndarray):
258        x = self._cache
259        grad_W = x.T @ grad_out
260        grad_b = grad_out.sum(axis=0) if self.b is not None else None
261        grad_x = grad_out @ self.W.T
262        return grad_x, grad_W, grad_b

269def psnr(img1: np.ndarray, img2: np.ndarray,
270         max_val: float = 255.0) -> float:
271    """
272    Peak Signal-to-Noise Ratio (dB).
273    Higher = better quality. Typical range: 30–50 dB for video.
274    PSNR = 10 * log10(MAX^2 / MSE)
275    """
276    mse = np.mean((img1.astype(float) - img2.astype(float)) ** 2)
277    if mse == 0:
278        return float('inf')
279    return 10 * np.log10(max_val ** 2 / mse)

282def ssim(img1: np.ndarray, img2: np.ndarray,
283         max_val: float = 255.0) -> float:
284    """
285    Structural Similarity Index (scalar approximation, no sliding window).
286    Compares luminance, contrast, and structure.
287    Range: [-1, 1], 1 = identical.
288 
289    Full SSIM uses local patches; this is the global version for simplicity.
290    """
291    c1 = (0.01 * max_val) ** 2
292    c2 = (0.03 * max_val) ** 2
293 
294    mu1, mu2 = img1.mean(), img2.mean()
295    sigma1_sq = img1.var()
296    sigma2_sq = img2.var()
297    sigma12 = np.mean((img1 - mu1) * (img2 - mu2))
298 
299    numerator   = (2 * mu1 * mu2 + c1) * (2 * sigma12 + c2)
300    denominator = (mu1**2 + mu2**2 + c1) * (sigma1_sq + sigma2_sq + c2)
301    return numerator / denominator

304def temporal_warping_error(frames_out: list, optical_flows: list) -> float:
305    """
306    Measures temporal consistency: how much the output changes relative
307    to what optical flow predicts between consecutive frames.
308 
309    warping_error_t = || output_t - warp(output_{t-1}, flow_{t-1->t}) ||^2
310 
311    frames_out:   list of output frames [T, H, W, C]
312    optical_flows: list of flows [T-1, H, W, 2] (flow[t] maps frame t to t+1)
313 
314    Note: real warp uses bilinear sampling. Here we use a simplified version
315    for illustration; in practice use cv2.remap or torch grid_sample.
316    """
317    from scipy.ndimage import map_coordinates
318 
319    total_error = 0.0
320    for t in range(1, len(frames_out)):
321        prev = frames_out[t - 1].astype(float)
322        curr = frames_out[t].astype(float)
323        flow = optical_flows[t - 1]             # [H, W, 2]
324        H, W = prev.shape[:2]
325 
326        grid_y, grid_x = np.mgrid[0:H, 0:W]
327        # flow gives displacement: new_x = x + flow_x
328        sample_y = grid_y - flow[..., 1]
329        sample_x = grid_x - flow[..., 0]
330 
331        warped = np.stack([
332            map_coordinates(prev[..., c], [sample_y, sample_x], order=1)
333            for c in range(prev.shape[2])
334        ], axis=-1)
335 
336        total_error += np.mean((curr - warped) ** 2)
337 
338    return total_error / (len(frames_out) - 1)

341def cosine_similarity(a: np.ndarray, b: np.ndarray) -> float:
342    """
343    Cosine similarity between two vectors.
344    Range: [-1, 1]. 1 = same direction, 0 = orthogonal.
345    Used in embedding comparison, feature matching.
346    """
347    norm_a = np.linalg.norm(a)
348    norm_b = np.linalg.norm(b)
349    if norm_a == 0 or norm_b == 0:
350        return 0.0
351    return float(np.dot(a, b) / (norm_a * norm_b))

358def knn_predict(X_train: np.ndarray, y_train: np.ndarray,
359                X_test: np.ndarray, k: int = 3) -> np.ndarray:
360    """
361    K-Nearest Neighbours (brute force, L2 distance).
362    X_train: [N, D], y_train: [N], X_test: [M, D]
363    Returns predicted labels [M].
364    """
365    predictions = []
366    for x in X_test:
367        dists = np.linalg.norm(X_train - x, axis=1)    # [N]
368        nn_idx = np.argsort(dists)[:k]
369        nn_labels = y_train[nn_idx]
370        # majority vote
371        counts = np.bincount(nn_labels.astype(int))
372        predictions.append(np.argmax(counts))
373    return np.array(predictions)

376def kmeans(X: np.ndarray, k: int, max_iter: int = 100,
377           seed: int = 0) -> tuple:
378    """
379    K-Means clustering.
380    Returns (centroids [k, D], labels [N]).
381    """
382    rng = np.random.default_rng(seed)
383    # random initialisation
384    centroids = X[rng.choice(len(X), k, replace=False)]
385 
386    for _ in range(max_iter):
387        # assign each point to nearest centroid
388        dists = np.linalg.norm(X[:, None] - centroids[None], axis=2)  # [N, k]
389        labels = np.argmin(dists, axis=1)                               # [N]
390 
391        new_centroids = np.array([
392            X[labels == i].mean(axis=0) if (labels == i).any() else centroids[i]
393            for i in range(k)
394        ])
395 
396        if np.allclose(centroids, new_centroids):
397            break
398        centroids = new_centroids
399 
400    return centroids, labels

407def confusion_matrix(y_true: np.ndarray, y_pred: np.ndarray,
408                     num_classes: int) -> np.ndarray:
409    """Returns [num_classes, num_classes] confusion matrix."""
410    cm = np.zeros((num_classes, num_classes), dtype=int)
411    for t, p in zip(y_true, y_pred):
412        cm[t][p] += 1
413    return cm

416def precision_recall_f1(y_true: np.ndarray, y_pred: np.ndarray,
417                         pos_label: int = 1) -> tuple:
418    """
419    Binary precision, recall, F1.
420    precision = TP / (TP + FP)  — of all predicted positive, how many correct
421    recall    = TP / (TP + FN)  — of all actual positive, how many caught
422    F1        = harmonic mean of precision and recall
423    """
424    tp = np.sum((y_pred == pos_label) & (y_true == pos_label))
425    fp = np.sum((y_pred == pos_label) & (y_true != pos_label))
426    fn = np.sum((y_pred != pos_label) & (y_true == pos_label))
427 
428    precision = tp / (tp + fp) if (tp + fp) > 0 else 0.0
429    recall    = tp / (tp + fn) if (tp + fn) > 0 else 0.0
430    f1 = (2 * precision * recall / (precision + recall)
431          if (precision + recall) > 0 else 0.0)
432    return precision, recall, f1

435def roc_auc(y_true: np.ndarray, y_scores: np.ndarray) -> tuple:
436    """
437    Compute ROC curve (fpr, tpr) and AUC via trapezoidal rule.
438    y_scores: predicted probability for the positive class.
439    """
440    thresholds = np.sort(np.unique(y_scores))[::-1]
441    fprs, tprs = [0.0], [0.0]
442 
443    total_pos = y_true.sum()
444    total_neg = len(y_true) - total_pos
445 
446    for thresh in thresholds:
447        pred = (y_scores >= thresh).astype(int)
448        tp = np.sum((pred == 1) & (y_true == 1))
449        fp = np.sum((pred == 1) & (y_true == 0))
450        fprs.append(fp / total_neg if total_neg > 0 else 0)
451        tprs.append(tp / total_pos if total_pos > 0 else 0)
452 
453    fprs.append(1.0)
454    tprs.append(1.0)
455    fprs, tprs = np.array(fprs), np.array(tprs)
456    auc = np.trapz(tprs, fprs)
457    return fprs, tprs, auc

464def min_max_normalize(x: np.ndarray) -> np.ndarray:
465    """Scale to [0, 1] per feature."""
466    x_min = x.min(axis=0)
467    x_max = x.max(axis=0)
468    denom = x_max - x_min
469    denom[denom == 0] = 1          # avoid divide-by-zero for constant features
470    return (x - x_min) / denom

473def z_score_normalize(x: np.ndarray) -> np.ndarray:
474    """Zero mean, unit variance per feature."""
475    mean = x.mean(axis=0)
476    std  = x.std(axis=0)
477    std[std == 0] = 1
478    return (x - mean) / std

481def shuffle_dataset(X: np.ndarray, y: np.ndarray,
482                    seed: int = 0) -> tuple:
483    """Shuffle X and y in unison without built-in shuffle."""
484    rng = np.random.default_rng(seed)
485    idx = rng.permutation(len(X))

489def train_val_split(X: np.ndarray, y: np.ndarray,
490                    val_ratio: float = 0.2, seed: int = 0) -> tuple:
491    """
492    Split dataset into train and validation sets.
493    For video data: split by clip/scene index, not by random frame,
494    to avoid temporal leakage between adjacent frames.
495    """
496    X, y = shuffle_dataset(X, y, seed)
497    n_val = int(len(X) * val_ratio)
498    return X[n_val:], y[n_val:], X[:n_val], y[:n_val]
