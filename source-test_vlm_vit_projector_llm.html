34def groups(model: VLM):
35    llm = [p for name, p in model.named_parameters() if not name.startswith(("vision.", "projector."))]
36    return list(model.vision.parameters()), list(model.projector.parameters()), llm

39class PatchTests(unittest.TestCase):
40    def setUp(self):
41        torch.manual_seed(0)

43    def test_patchify_matches_manual_slicing_in_row_major_order(self):
44        images = torch.randn(2, 3, 8, 8)
45        patches = patchify(images, 4)
46        self.assertEqual(tuple(patches.shape), (2, 4, 48))
47        for b in range(2):
48            for r in range(2):
49                for c in range(2):
50                    block = images[b, :, 4 * r:4 * r + 4, 4 * c:4 * c + 4]          # [C, P, P]
51                    torch.testing.assert_close(patches[b, 2 * r + c], block.permute(1, 2, 0).reshape(-1))
52 
53    def test_patch_embedding_equals_a_convolution_with_kernel_and_stride_p(self):
54        embed = PatchEmbedding(image_size=8, patch=4, channels=3, d_vision=8)
55        conv = nn.Conv2d(3, 8, kernel_size=4, stride=4)
56        with torch.no_grad():
57            conv.weight.copy_(embed.linear.weight.view(8, 4, 4, 3).permute(0, 3, 1, 2))
58            conv.bias.copy_(embed.linear.bias)
59        images = torch.randn(2, 3, 8, 8)
60        expected = conv(images).flatten(2).transpose(1, 2) + embed.positions        # [B, N, d_v]
61        torch.testing.assert_close(embed(images), expected, atol=1e-6, rtol=1e-5)

84    def test_splice_puts_the_image_tokens_where_the_placeholder_stands(self):
85        t = self.text_ids[:1]
86        U = self.model.encode_image(self.images[:1])
87        X0 = self.model.splice(t, U) - self.model.positions[:11]
88        E = self.model.embedding(t)
89        torch.testing.assert_close(X0[0, 0], E[0, 0])          # BOS
90        torch.testing.assert_close(X0[0, 1:5], U[0])           # the four image tokens
91        torch.testing.assert_close(X0[0, 5:], E[0, 2:])        # what is this ? a cat
92        self.assertFalse(torch.allclose(X0[0, 1], E[0, 1]))    # the <image> row itself is never used

100    def test_llm_is_causal_and_text_reads_the_image_tokens(self):
101        t = self.text_ids[:1]
102        other = t.clone()
103        other[0, 6:] = torch.tensor([11, 12])                  # change the two tokens after '?'
104        base = self.model(self.images[:1], t)
105        torch.testing.assert_close(self.model(self.images[:1], other)[:, :9], base[:, :9])
106        with_dog = self.model(self.images[1:], t)              # same text, other image
107        torch.testing.assert_close(with_dog[:, 0], base[:, 0])          # BOS precedes the image
108        self.assertFalse(torch.allclose(with_dog[:, 5:], base[:, 5:]))  # every word after it changed

110    def test_only_the_answer_positions_are_trained_on(self):
111        labels = self.model.splice_labels(self.text_ids, self.text_labels, 4)
112        self.assertEqual(labels[0].tolist(), [IGNORE] * 8 + [8, 9, 2])
113        self.assertEqual(labels[1].tolist(), [IGNORE] * 8 + [8, 10, 2])
114        logits = self.model(self.images, self.text_ids)
115        log_p = F.log_softmax(logits, dim=-1)
116        manual = -torch.stack([log_p[b, j, labels[b, j]] for b in range(2) for j in (8, 9, 10)]).mean()
117        torch.testing.assert_close(answer_loss(logits, labels), manual)
118        noisy = logits.clone()
119        noisy[:, :8] += torch.randn_like(noisy[:, :8])         # image and question positions
120        torch.testing.assert_close(answer_loss(noisy, labels), manual)

122    def test_each_stage_updates_only_its_own_parameters(self):
123        labels = self.model.splice_labels(self.text_ids, self.text_labels, 4)
124        vision, projector, llm = groups(self.model)
125        expected = {0: (False, False, True), 1: (False, True, False), 2: (False, True, True)}
126        for stage, trains in expected.items():
127            with self.subTest(stage=stage):
128                set_stage(self.model, stage)
129                self.model.zero_grad()
130                for parameter in self.model.parameters():
131                    parameter.grad = None
132                answer_loss(self.model(self.images, self.text_ids), labels).backward()
133                for group, trained in zip((vision, projector, llm), trains):
134                    self.assertTrue(all((p.grad is not None) == trained for p in group))
135                if trains[1]:   # the gradient reached the projector through the LLM layers
136                    self.assertTrue(any(p.grad.abs().sum() > 0 for p in projector))

149class TrainingTests(unittest.TestCase):
150    def test_alignment_lets_a_frozen_llm_read_the_image(self):
151        torch.manual_seed(0)
152        images, text_ids, text_labels, captions = toy_batch()
153        model = make_model()
154        labels = model.splice_labels(text_ids, text_labels, 4)
155        vision, projector, llm = groups(model)
156        snapshot = lambda group: [p.detach().clone() for p in group]
157        same = lambda group, saved: all(torch.equal(p, q) for p, q in zip(group, saved))
158        vision_start = snapshot(vision)
159 
160        # Stage 0: language only. The LLM learns to read the answer from words in the image slot.
161        _, _, after = train(model, 0, lambda: model(None, text_ids, model.embedding(captions)), labels, 300)
162        self.assertLess(after, 0.05)
163        llm_pretrained, projector_start = snapshot(llm), snapshot(projector)
164        self.assertTrue(same(projector, projector_start))
165 
166        # Stage 1: the projector alone makes the image readable for the frozen LLM.
167        count, before, after = train(model, 1, lambda: model(images, text_ids), labels, 300)
168        self.assertEqual(count, sum(p.numel() for p in projector))
169        self.assertGreater(before, 0.3)
170        self.assertLess(after, 0.05)
171        self.assertTrue(same(llm, llm_pretrained))
172        self.assertFalse(same(projector, projector_start))
173        model.eval()
174        self.assertEqual(generate(model, images, text_ids[:, :6]).tolist(), [CAT_ANSWER, DOG_ANSWER])
175        swapped = generate(model, images.flip(0), text_ids[:, :6]).tolist()
176        self.assertEqual(swapped, [DOG_ANSWER, CAT_ANSWER])     # the answer follows the image, not the row
177 
178        # Stage 2: projector and LLM together; the vision encoder never moved.
179        model.train()
180        _, _, after = train(model, 2, lambda: model(images, text_ids), labels, 100)
181        self.assertLess(after, 0.01)
182        self.assertFalse(same(llm, llm_pretrained))
183        self.assertTrue(same(vision, vision_start))
