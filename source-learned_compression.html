1"""
2Interview-style learned image compression (end-to-end) using PyTorch.
3This is a minimal, illustrative pipeline (not a full paper implementation).
4"""
5 
6import torch
7import torch.nn as nn
8import torch.nn.functional as F
9 
10 
11class Encoder(nn.Module):
12    def __init__(self, in_ch=3, latent_ch=64):
13        super().__init__()
14        self.net = nn.Sequential(
15            nn.Conv2d(in_ch, 64, 5, stride=2, padding=2),
16            nn.ReLU(inplace=True),
17            nn.Conv2d(64, 128, 5, stride=2, padding=2),
18            nn.ReLU(inplace=True),
19            nn.Conv2d(128, latent_ch, 5, stride=2, padding=2),
20        )
21 
22    def forward(self, x):
23        return self.net(x)
24 
25 
26class Decoder(nn.Module):
27    def __init__(self, latent_ch=64, out_ch=3):
28        super().__init__()
29        self.net = nn.Sequential(
30            nn.ConvTranspose2d(latent_ch, 128, 5, stride=2, padding=2, output_padding=1),
31            nn.ReLU(inplace=True),
32            nn.ConvTranspose2d(128, 64, 5, stride=2, padding=2, output_padding=1),
33            nn.ReLU(inplace=True),
34            nn.ConvTranspose2d(64, out_ch, 5, stride=2, padding=2, output_padding=1),
35            nn.Sigmoid(),
36        )
37 
38    def forward(self, y):
39        return self.net(y)
40 
41 
42class EntropyModel(nn.Module):
43    """
44    Simplified factorized Gaussian likelihood model.
45    Returns per-latent probability (likelihood).
46    """
47 
48    def __init__(self, latent_ch=64):
49        super().__init__()
50        self.log_scale = nn.Parameter(torch.zeros(latent_ch, 1, 1))
51 
52    def forward(self, y_q):
53        scale = torch.exp(self.log_scale) + 1e-6
54        # Gaussian PDF for each latent element (factorized).
55        likelihood = (1.0 / (torch.sqrt(torch.tensor(2.0 * torch.pi)) * scale)) * torch.exp(
56            -0.5 * (y_q / scale) ** 2
57        )
58        return torch.clamp(likelihood, min=1e-9)
59class LearnedCompressionModel(nn.Module):
60    def __init__(self, in_ch=3, latent_ch=64):
61        super().__init__()
62        self.g_a = Encoder(in_ch, latent_ch)
63        self.g_s = Decoder(latent_ch, in_ch)
64        self.entropy_model = EntropyModel(latent_ch)
65    def forward(self, x):
66        y = self.g_a(x)
67        if self.training:
68            y_q = y + torch.empty_like(y).uniform_(-0.5, 0.5)
69        else:
70            y_q = torch.round(y)
71        x_hat = self.g_s(y_q)
72        likelihood = self.entropy_model(y_q)
73        return x_hat, y_q, likelihood
74def rd_loss(x, x_hat, likelihood, lam=0.01):
75    bpp = (-torch.log2(likelihood)).mean()
76    dist = torch.mean((x - x_hat) ** 2)
77    return bpp + lam * dist, bpp, dist
78def train_step(model, x, optimizer, lam=0.01):
79    model.train()
80    x_hat, y_q, likelihood = model(x)
81    loss, bpp, dist = rd_loss(x, x_hat, likelihood, lam=lam)
82    optimizer.zero_grad()
83    loss.backward()
84    optimizer.step()
85    return {"loss": float(loss.item()), "bpp": float(bpp.item()), "dist": float(dist.item())}
86@torch.no_grad()
87def inference(model, x):
88    model.eval()
89    x_hat, y_q, likelihood = model(x)
90    bpp = (-torch.log2(likelihood)).mean()
91    return x_hat, bpp
92if __name__ == "__main__":
93    B, H, W = 2, 128, 128
94    x = torch.rand(B, 3, H, W)
95    model = LearnedCompressionModel(in_ch=3, latent_ch=64)
96    optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)
97    stats = train_step(model, x, optimizer, lam=0.01)
98    print("train:", stats)
99    x_hat, bpp = inference(model, x)
100    print("inference bpp:", float(bpp.item()))
