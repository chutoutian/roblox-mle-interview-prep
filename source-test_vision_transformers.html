111def reference_window_mix(block: SwinBlock, h, H, W):
112    """(S)W-MSA without roll or window_partition: global attention over all H*W tokens where a
113    pair may attend only if both tokens lie in the same (shifted) window of the image, and the
114    bias is looked up from their true offset. Window of row y: floor((y - s) / M)."""
115    M = block.M
116    s = block.shift if min(H, W) > M else 0
117    ys, xs = torch.meshgrid(torch.arange(H), torch.arange(W), indexing="ij")
118    y, x = ys.flatten(), xs.flatten()
119    wy, wx = torch.div(y - s, M, rounding_mode="floor"), torch.div(x - s, M, rounding_mode="floor")
120    same = (wy[:, None] == wy[None, :]) & (wx[:, None] == wx[None, :])
121    dy = (y[:, None] - y[None, :]).clamp(1 - M, M - 1) + M - 1                       # clamped pairs are masked
122    dx = (x[:, None] - x[None, :]).clamp(1 - M, M - 1) + M - 1
123    bias = block.table[:, dy * (2 * M - 1) + dx]                                       # [heads, N, N]
124    return block.attn(h, bias=bias, mask=same[None, None])

240class DINOTests(unittest.TestCase):
241    def test_loss_formula_and_centre(self):
242        torch.manual_seed(0)
243        student = [torch.randn(3, 5) for _ in range(4)]                                # 2 global + 2 local crops
244        teacher = [torch.randn(3, 5) for _ in range(2)]
245        loss = DINOLoss(5, tau_s=0.1, tau_t=0.04, center_momentum=0.9)
246        value = loss(student, teacher)
247        terms = []
248        for i, t in enumerate(teacher):
249            p_t = F.softmax(t / 0.04, dim=-1)                                          # centre starts at 0
250            for j, s in enumerate(student):
251                if j != i:
252                    terms.append(-(p_t * F.log_softmax(s / 0.1, dim=-1)).sum(-1).mean())
253        self.assertEqual(len(terms), 2 * 3)
254        self.assertTrue(close(value, torch.stack(terms).mean()))
255        self.assertTrue(close(loss.center, 0.1 * torch.cat(teacher).mean(0, keepdim=True)))
256        c = torch.randn(1, 5)
257        loss.center.copy_(c)                                                           # a non-zero centre is subtracted
258        value = loss(student, teacher)
259        terms = [-(F.softmax((t - c) / 0.04, dim=-1) * F.log_softmax(s / 0.1, dim=-1)).sum(-1).mean()
260                 for i, t in enumerate(teacher) for j, s in enumerate(student) if j != i]
261        self.assertTrue(close(value, torch.stack(terms).mean()))
262        self.assertTrue(close(loss.center, 0.9 * c + 0.1 * torch.cat(teacher).mean(0, keepdim=True)))

264    def test_only_the_student_gets_gradients_and_the_teacher_is_an_ema(self):
265        torch.manual_seed(0)

338    def test_adaln_zero_modulation(self):
339        torch.manual_seed(0)
340        D = 8
341        block = DiTBlock(dim=D, heads=2)
342        with torch.no_grad():
343            block.ada[1].weight.normal_()
344            block.ada[1].bias.normal_()                                                # leave the zero start
345        x, c = torch.randn(2, 16, D), torch.randn(2, D)
346        p = block.ada(c)[:, None]                                                      # DiT order: shift, scale, gate
347        beta1, gamma1, alpha1 = p[..., :D], p[..., D:2 * D], p[..., 2 * D:3 * D]
348        beta2, gamma2, alpha2 = p[..., 3 * D:4 * D], p[..., 4 * D:5 * D], p[..., 5 * D:]
349        ln = lambda t: F.layer_norm(t, (D,), eps=1e-6)
350        h = x + alpha1 * block.attn((1 + gamma1) * ln(x) + beta1)
351        self.assertTrue(close(block(x, c), h + alpha2 * block.mlp((1 + gamma2) * ln(h) + beta2)))
