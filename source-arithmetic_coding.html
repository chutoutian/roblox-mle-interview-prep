5def _build_model(freqs):
6    """Return sorted symbols, cumulative counts, and total count."""
7    symbols = sorted(freqs)
8    cum = [0]
9    for s in symbols:
10        c = int(freqs[s])
11        if c <= 0:
12            raise ValueError("All frequencies must be positive.")
13        cum.append(cum[-1] + c)
14    total = cum[-1]
15    if total <= 0:
16        raise ValueError("Total frequency must be positive.")
17    return symbols, cum, total

20def arithmetic_encode(data, freqs=None, precision=32):
21    """
22    Static arithmetic encoder (integer range coder style).
23    Returns: (bitstream, freqs)
24    """
25    if precision < 16:
26        raise ValueError("precision must be >= 16")
27    if not data:
28        return "", (dict(freqs) if freqs is not None else {})
29 
30    if freqs is None:
31        freqs = dict(Counter(data))
32 
33    symbols, cum, total = _build_model(freqs)
34    sym_to_idx = {s: i for i, s in enumerate(symbols)}
35 
36    FULL = (1 << precision) - 1
37    HALF = 1 << (precision - 1)
38    QTR = 1 << (precision - 2)
39    THREE_QTR = 3 * QTR
40 
41    low, high = 0, FULL
42    pending = 0
43    bits = []
44 
45    def emit(bit):
46        nonlocal pending
47        bits.append("1" if bit else "0")
48        # flush deferred opposite bits (E3 handling)
49        fill = "0" if bit else "1"
50        while pending > 0:
51            bits.append(fill)
52            pending -= 1
53 
54    for s in data:
55        i = sym_to_idx[s]
56        s_low, s_high = cum[i], cum[i + 1]
57 
58        rng = high - low + 1
59        high = low + (rng * s_high // total) - 1
60        low = low + (rng * s_low // total)
61 
62        while True:
63            if high < HALF:               # E1
64                emit(0)
65            elif low >= HALF:            # E2
66                emit(1)
67                low -= HALF
68                high -= HALF
69            elif low >= QTR and high < THREE_QTR:  # E3
70                pending += 1
71                low -= QTR
72                high -= QTR
73            else:
74                break
75 
76            low = low << 1
77            high = (high << 1) + 1
78 
79    # finalize
80    pending += 1
81    emit(0 if low < QTR else 1)
82 
83    return "".join(bits), freqs

86def arithmetic_decode(bitstream, freqs, n_symbols, precision=32):
87    """
88    Static arithmetic decoder matching arithmetic_encode.
89    """
90    if n_symbols == 0:
91        return []
92    if precision < 16:
93        raise ValueError("precision must be >= 16")
94 
95    symbols, cum, total = _build_model(freqs)
96 
97    FULL = (1 << precision) - 1
98    HALF = 1 << (precision - 1)
99    QTR = 1 << (precision - 2)
100    THREE_QTR = 3 * QTR
101 
102    idx = 0
103 
104    def read_bit():
105        nonlocal idx
106        if idx < len(bitstream):
107            b = bitstream[idx]
108            idx += 1
109            if b == "0":
110                return 0
111            if b == "1":
112                return 1
113            raise ValueError("bitstream must contain only '0'/'1'")
114        return 0  # implicit zero padding
115 
116    low, high = 0, FULL
117    value = 0
118    for _ in range(precision):
119        value = (value << 1) | read_bit()
120 
121    out = []
122    for _ in range(n_symbols):
123        rng = high - low + 1
124        scaled = ((value - low + 1) * total - 1) // rng
125 
126        # find i s.t. cum[i] <= scaled < cum[i+1]
127        i = bisect_right(cum, scaled) - 1
128        out.append(symbols[i])
129 
130        s_low, s_high = cum[i], cum[i + 1]
131        high = low + (rng * s_high // total) - 1
132        low = low + (rng * s_low // total)
133 
134        while True:
135            if high < HALF:               # E1
136                pass
137            elif low >= HALF:            # E2
138                low -= HALF
139                high -= HALF
140                value -= HALF
141            elif low >= QTR and high < THREE_QTR:  # E3
142                low -= QTR
143                high -= QTR
144                value -= QTR
145            else:
146                break
147 
148            low = low << 1
149            high = (high << 1) + 1
150            value = (value << 1) | read_bit()
151 
152    return out
