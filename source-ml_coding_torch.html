38def binary_cross_entropy(pred: Tensor, target: Tensor,
39                         eps: float = 1e-9) -> Tensor:
40    """pred: probabilities in (0,1), target: {0,1}"""
41    pred = pred.clamp(eps, 1 - eps)
42    return -(target * pred.log() + (1 - target) * (1 - pred).log()).mean()

45def cross_entropy_loss(logits: Tensor, labels: Tensor) -> Tensor:
46    """
47    From raw logits — applies log-softmax internally (numerically stable).
48    logits: [N, C],  labels: [N] integer class indices.
49    """
50    # log-sum-exp trick for numerical stability
51    shifted = logits - logits.max(dim=1, keepdim=True).values
52    log_sum_exp = shifted.exp().sum(dim=1, keepdim=True).log()
53    log_probs = shifted - log_sum_exp                       # [N, C]
54    return -log_probs[torch.arange(len(labels)), labels].mean()

57def rate_distortion_loss(distortion: Tensor, rate: Tensor,
58                         lam: float) -> Tensor:
59    """
60    Standard learned compression objective.
61    distortion: reconstruction error (e.g. MSE)
62    rate:       estimated bits (entropy of quantised latent)
63    lam:        larger → penalise bits more → lower bitrate
64    """
65    return distortion + lam * rate

72class SigmoidFunction(torch.autograd.Function):
73    """Sigmoid with manual backward — useful to show in interviews."""
74    @staticmethod
75    def forward(ctx, x: Tensor) -> Tensor:
76        s = 1 / (1 + (-x).exp())
77        ctx.save_for_backward(s)
78        return s
79 
80    @staticmethod
81    def backward(ctx, grad_out: Tensor):
82        (s,) = ctx.saved_tensors
83        return grad_out * s * (1 - s)
84 
85 
86sigmoid = SigmoidFunction.apply

89class ReLUFunction(torch.autograd.Function):
90    @staticmethod
91    def forward(ctx, x: Tensor) -> Tensor:
92        ctx.save_for_backward(x)
93        return x.clamp(min=0)
94 
95    @staticmethod
96    def backward(ctx, grad_out: Tensor):
97        (x,) = ctx.saved_tensors
98        return grad_out * (x > 0).float()

101relu = ReLUFunction.apply

104def softmax(x: Tensor, dim: int = -1) -> Tensor:
105    """Numerically stable softmax."""
106    shifted = x - x.max(dim=dim, keepdim=True).values
107    exp_x = shifted.exp()
108    return exp_x / exp_x.sum(dim=dim, keepdim=True)

111def gelu(x: Tensor) -> Tensor:
112    """Gaussian Error Linear Unit — standard in transformers."""
113    return 0.5 * x * (1 + torch.tanh((2 / torch.pi) ** 0.5
114                                      * (x + 0.044715 * x ** 3)))

121class SGD:
122    def __init__(self, params: list, lr: float = 0.01,
123                 momentum: float = 0.0):
124        self.params = params
125        self.lr = lr
126        self.momentum = momentum
127        self.velocity = [torch.zeros_like(p) for p in params]
128 
129    @torch.no_grad()
130    def step(self):
131        for p, v in zip(self.params, self.velocity):
132            if p.grad is None:
133                continue
134            v.mul_(self.momentum).add_(p.grad, alpha=-self.lr)
135            p.add_(v)
136 
137    def zero_grad(self):
138        for p in self.params:
139            if p.grad is not None:
140                p.grad.zero_()

143class Adam:
144    """
145    Adam: adaptive per-parameter learning rates.
146    m = first moment (gradient mean)
147    v = second moment (gradient variance)
148    Bias-corrected estimates are used for the update.
149    """
150    def __init__(self, params: list, lr: float = 1e-3,
151                 beta1: float = 0.9, beta2: float = 0.999,
152                 eps: float = 1e-8, weight_decay: float = 0.0):
153        self.params = params
154        self.lr = lr
155        self.beta1 = beta1
156        self.beta2 = beta2
157        self.eps = eps
158        self.weight_decay = weight_decay
159        self.m = [torch.zeros_like(p) for p in params]
160        self.v = [torch.zeros_like(p) for p in params]
161        self.t = 0
162 
163    @torch.no_grad()
164    def step(self):
165        self.t += 1
166        for p, m, v in zip(self.params, self.m, self.v):
167            if p.grad is None:
168                continue
169            g = p.grad
170            if self.weight_decay > 0:
171                g = g + self.weight_decay * p          # AdamW-style decoupling
172 
173            m.mul_(self.beta1).add_(g, alpha=1 - self.beta1)
174            v.mul_(self.beta2).addcmul_(g, g, value=1 - self.beta2)
175 
176            m_hat = m / (1 - self.beta1 ** self.t)
177            v_hat = v / (1 - self.beta2 ** self.t)
178 
179            p.sub_(self.lr * m_hat / (v_hat.sqrt() + self.eps))
180 
181    def zero_grad(self):
182        for p in self.params:
183            if p.grad is not None:
184                p.grad.zero_()

191class BatchNorm1d(nn.Module):
192    """
193    BatchNorm for [N, C]: normalises each feature across the batch.
194    Maintains running mean/var for inference (no batch dependency at test time).
195    """
196    def __init__(self, num_features: int, eps: float = 1e-5,
197                 momentum: float = 0.1):
198        super().__init__()
199        self.eps = eps
200        self.momentum = momentum
201        self.gamma = nn.Parameter(torch.ones(num_features))
202        self.beta  = nn.Parameter(torch.zeros(num_features))
203        self.register_buffer('running_mean', torch.zeros(num_features))
204        self.register_buffer('running_var',  torch.ones(num_features))
205 
206    def forward(self, x: Tensor) -> Tensor:
207        """x: [N, C]"""
208        if self.training:
209            mean = x.mean(dim=0)
210            var  = x.var(dim=0, unbiased=False)
211            self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * mean
212            self.running_var  = (1 - self.momentum) * self.running_var  + self.momentum * var
213        else:
214            mean, var = self.running_mean, self.running_var
215 
216        x_hat = (x - mean) / (var + self.eps).sqrt()
217        return self.gamma * x_hat + self.beta

220class LayerNorm(nn.Module):
221    """
222    LayerNorm: normalises each sample across its feature dimension.
223    No batch dependency — identical behaviour at train and inference.
224    """
225    def __init__(self, normalized_shape: int, eps: float = 1e-5):
226        super().__init__()
227        self.eps = eps
228        self.gamma = nn.Parameter(torch.ones(normalized_shape))
229        self.beta  = nn.Parameter(torch.zeros(normalized_shape))
230 
231    def forward(self, x: Tensor) -> Tensor:
232        """x: [..., C] — normalises over last dimension."""
233        mean = x.mean(dim=-1, keepdim=True)
234        var  = x.var(dim=-1, keepdim=True, unbiased=False)
235        x_hat = (x - mean) / (var + self.eps).sqrt()
236        return self.gamma * x_hat + self.beta

239class ScaledDotProductAttention(nn.Module):
240    """
241    Attention(Q, K, V) = softmax(Q K^T / sqrt(d_k)) V
242    """
243    def __init__(self, dropout: float = 0.0):
244        super().__init__()
245        self.drop = nn.Dropout(dropout)
246 
247    def forward(self, Q: Tensor, K: Tensor, V: Tensor,
248                mask: Tensor = None) -> Tensor:
249        """
250        Q, K: [..., seq, d_k]
251        V:    [..., seq, d_v]
252        mask: additive mask (e.g. -inf for padding)
253        """
254        d_k = Q.size(-1)
255        scores = Q @ K.transpose(-2, -1) / d_k ** 0.5  # [..., seq, seq]
256        if mask is not None:
257            scores = scores + mask
258        weights = self.drop(softmax(scores, dim=-1))
259        return weights @ V                               # [..., seq, d_v]

262class MultiHeadAttention(nn.Module):
263    """
264    Multi-head self/cross attention.
265    Splits d_model into h heads of size d_k = d_model // h.
266    """
267    def __init__(self, d_model: int, num_heads: int, dropout: float = 0.0):
268        super().__init__()
269        assert d_model % num_heads == 0
270        self.h = num_heads
271        self.d_k = d_model // num_heads
272        self.W_q = nn.Linear(d_model, d_model)
273        self.W_k = nn.Linear(d_model, d_model)
274        self.W_v = nn.Linear(d_model, d_model)
275        self.W_o = nn.Linear(d_model, d_model)
276        self.attn = ScaledDotProductAttention(dropout)
277 
278    def _split_heads(self, x: Tensor) -> Tensor:
279        """[B, S, d_model] → [B, h, S, d_k]"""
280        B, S, _ = x.shape
281        return x.view(B, S, self.h, self.d_k).transpose(1, 2)
282 
283    def forward(self, Q: Tensor, K: Tensor, V: Tensor,
284                mask: Tensor = None) -> Tensor:
285        B = Q.size(0)
286        Q = self._split_heads(self.W_q(Q))
287        K = self._split_heads(self.W_k(K))
288        V = self._split_heads(self.W_v(V))
289        out = self.attn(Q, K, V, mask)                  # [B, h, S, d_k]
290        out = out.transpose(1, 2).reshape(B, -1, self.h * self.d_k)
291        return self.W_o(out)

294class Linear(nn.Module):
295    """Fully connected layer with He initialisation."""
296    def __init__(self, in_features: int, out_features: int,
297                 bias: bool = True):

300            torch.randn(out_features, in_features) * (2 / in_features) ** 0.5

312class LoRALinear(nn.Module):
313    """
314    LoRA wrapper around a frozen Linear layer.
315 
316    Forward:  h = W·x + (B·A)·x  =  (W + B·A)·x
317    Only A and B are trained; W is frozen.
318 
319    Merge at inference:  call .merge() to fold B·A into W,
320    making the layer identical in cost to a plain Linear.
321    """
322    def __init__(self, in_features: int, out_features: int,
323                 rank: int = 4, alpha: float = 1.0, bias: bool = True):
324        super().__init__()
325        self.rank = rank
326        self.scale = alpha / rank               # ΔW is multiplied by scale
327 
328        # frozen base weight
329        self.weight = nn.Parameter(
330            torch.randn(out_features, in_features) * (2 / in_features) ** 0.5,
331            requires_grad=False,
332        )
333        self.bias = nn.Parameter(torch.zeros(out_features)) if bias else None
334 
335        # trainable low-rank matrices
336        # A: Gaussian init, B: zero init → ΔW = B·A = 0 at start
337        self.A = nn.Parameter(torch.randn(rank, in_features) * 0.01)
338        self.B = nn.Parameter(torch.zeros(out_features, rank))
339 
340        self._merged = False
341 
342    def forward(self, x: Tensor) -> Tensor:
343        if self._merged:
344            return F.linear(x, self.weight, self.bias)
345        base = F.linear(x, self.weight, self.bias)
346        lora = F.linear(F.linear(x, self.A), self.B) * self.scale
347        return base + lora
348 
349    def merge(self):
350        """
351        Fold ΔW = scale * B·A into W (in-place).
352        After merging: zero inference overhead, identical to plain Linear.
353        """
354        if self._merged:
355            return
356        with torch.no_grad():
357            self.weight.add_(self.scale * self.B @ self.A)
358        self._merged = True
359 
360    def unmerge(self):
361        if not self._merged:
362            return
363        with torch.no_grad():
364            self.weight.sub_(self.scale * self.B @ self.A)
365        self._merged = False
366 
367    def trainable_params(self) -> int:
368        return self.A.numel() + self.B.numel()
369 
370    def total_params(self) -> int:
371        return self.weight.numel() + self.trainable_params()

378def psnr(pred: Tensor, target: Tensor, max_val: float = 1.0) -> Tensor:
379    """
380    Peak Signal-to-Noise Ratio (dB). Higher = better.
381    pred, target: any shape, values in [0, max_val].
382    """
383    mse = ((pred - target) ** 2).mean()
384    if mse == 0:
385        return torch.tensor(float('inf'))
386    return 10 * torch.log10(max_val ** 2 / mse)

389def ssim(pred: Tensor, target: Tensor, max_val: float = 1.0) -> Tensor:
390    """
391    Global SSIM (no sliding window — for illustration).
392    pred, target: [N, C, H, W] or flattened.
393    """
394    c1 = (0.01 * max_val) ** 2
395    c2 = (0.03 * max_val) ** 2
396 
397    mu1, mu2 = pred.mean(), target.mean()
398    sigma1_sq = pred.var()
399    sigma2_sq = target.var()
400    sigma12   = ((pred - mu1) * (target - mu2)).mean()
401 
402    num = (2 * mu1 * mu2 + c1) * (2 * sigma12 + c2)
403    den = (mu1**2 + mu2**2 + c1) * (sigma1_sq + sigma2_sq + c2)
404    return num / den

407def cosine_similarity(a: Tensor, b: Tensor, eps: float = 1e-8) -> Tensor:
408    """Cosine similarity between two vectors (1D tensors)."""
409    return torch.dot(a, b) / (a.norm() * b.norm() + eps)

412def temporal_warping_error(frames: Tensor, flows: Tensor) -> Tensor:
413    """
414    Temporal consistency metric.
415    frames: [T, C, H, W]
416    flows:  [T-1, 2, H, W]  (flow[t] warps frame t to t+1)
417    Returns mean squared warping error across time.
418    """
419    T, C, H, W = frames.shape
420    total = torch.tensor(0.0)
421 
422    for t in range(T - 1):
423        flow = flows[t]                                   # [2, H, W]
424        prev = frames[t].unsqueeze(0)                     # [1, C, H, W]
425        curr = frames[t + 1]
426 
427        # build sampling grid
428        grid_y, grid_x = torch.meshgrid(
429            torch.linspace(-1, 1, H),
430            torch.linspace(-1, 1, W),
431            indexing='ij',
432        )
433        # normalise flow to [-1, 1] range
434        dx = flow[0] / (W / 2)
435        dy = flow[1] / (H / 2)
436        grid = torch.stack([grid_x + dx, grid_y + dy], dim=-1).unsqueeze(0)
437 
438        warped = F.grid_sample(prev, grid, align_corners=True,
439                               mode='bilinear', padding_mode='border')
440        total = total + ((curr - warped.squeeze(0)) ** 2).mean()
441 
442    return total / (T - 1)

449def knn_predict(X_train: Tensor, y_train: Tensor,
450                X_test: Tensor, k: int = 3) -> Tensor:
451    """
452    K-Nearest Neighbours (brute-force L2).
453    X_train: [N, D],  y_train: [N],  X_test: [M, D]
454    Returns predicted labels [M].
455    """
456    # pairwise L2: ||a - b||^2 = ||a||^2 + ||b||^2 - 2 a·b^T
457    a_sq = (X_test ** 2).sum(dim=1, keepdim=True)           # [M, 1]
458    b_sq = (X_train ** 2).sum(dim=1, keepdim=True).T        # [1, N]
459    dists = (a_sq + b_sq - 2 * X_test @ X_train.T).clamp(min=0).sqrt()
460 
461    nn_idx  = dists.topk(k, dim=1, largest=False).indices   # [M, k]
462    nn_lbls = y_train[nn_idx]                               # [M, k]
463    # majority vote
464    predictions = nn_lbls.mode(dim=1).values
465    return predictions

468def kmeans(X: Tensor, k: int, max_iter: int = 100) -> tuple:
469    """
470    K-Means clustering.
471    Returns (centroids [k, D], labels [N]).
472    """
473    idx = torch.randperm(len(X))[:k]
474    centroids = X[idx].clone()
475 
476    for _ in range(max_iter):
477        # [N, k] pairwise distances
478        dists = torch.cdist(X, centroids)
479        labels = dists.argmin(dim=1)                        # [N]
480 
481        new_centroids = torch.stack([
482            X[labels == i].mean(dim=0) if (labels == i).any()
483            else centroids[i]
484            for i in range(k)
485        ])
486 
487        if torch.allclose(centroids, new_centroids):
488            break
489        centroids = new_centroids
490 
491    return centroids, labels

498def confusion_matrix(y_true: Tensor, y_pred: Tensor,
499                     num_classes: int) -> Tensor:
500    cm = torch.zeros(num_classes, num_classes, dtype=torch.long)
501    for t, p in zip(y_true, y_pred):
502        cm[t, p] += 1
503    return cm

506def precision_recall_f1(y_true: Tensor, y_pred: Tensor,
507                         pos_label: int = 1) -> tuple:
508    tp = ((y_pred == pos_label) & (y_true == pos_label)).sum().float()
509    fp = ((y_pred == pos_label) & (y_true != pos_label)).sum().float()
510    fn = ((y_pred != pos_label) & (y_true == pos_label)).sum().float()
511 
512    precision = tp / (tp + fp + 1e-9)
513    recall    = tp / (tp + fn + 1e-9)
514    f1 = 2 * precision * recall / (precision + recall + 1e-9)
515    return precision, recall, f1

522def min_max_normalize(x: Tensor) -> Tensor:
523    x_min = x.min(dim=0).values
524    x_max = x.max(dim=0).values
525    denom = (x_max - x_min).clamp(min=1e-9)
526    return (x - x_min) / denom

529def z_score_normalize(x: Tensor) -> Tensor:
530    mean = x.mean(dim=0)
531    std  = x.std(dim=0).clamp(min=1e-9)
532    return (x - mean) / std
