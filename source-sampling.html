45def upsample(arr):
46    m = len(arr)
47    n = len(arr[0])
48 
49    rgb = [[[0, 0, 0] for _ in range(n*2)] for _ in range(m*2)]
50 
51    for i in range(m):
52        for j in range(n):
53            y, u, v = arr[i][j]
54 
55            r = y + 1.13983 * v
56            g = y - 0.39465 * u - 0.58060 * v
57            b = y + 2.03211 * u
58 
59            rgb[i*2][j*2] = [r, g, b]
60            rgb[i*2][j*2+1] = [r, g, b]
61            rgb[i*2+1][j*2] = [r, g, b]
62            rgb[i*2+1][j*2+1] = [r, g, b]
63 
64    return rgb
65def upsample_bilinear_2x(arr):
66    m = len(arr)
67    if m == 0:
68        return []
69    n = len(arr[0])
70    if n == 0:
71        return [[]]
72 
73    out_h, out_w = m * 2, n * 2
74    out = [[[0.0, 0.0, 0.0] for _ in range(out_w)] for _ in range(out_h)]
75 
76    # 
77    for i2 in range(out_h):
78        # Map output y to source y
79        y = i2 / 2.0
80        y0 = int(y)
81        y1 = min(y0 + 1, m - 1)
82        wy = y - y0  # in [0,1]
83 
84        for j2 in range(out_w):
85            # Map output x to source x
86            x = j2 / 2.0
87            x0 = int(x)
88            x1 = min(x0 + 1, n - 1)
89            wx = x - x0  # in [0,1]
90 
91            # Bilinear for each channel
92            for c in range(3):
93                v00 = arr[y0][x0][c]
94                v01 = arr[y0][x1][c]
95                v10 = arr[y1][x0][c]
96                v11 = arr[y1][x1][c]
97 
98                top = v00 * (1.0 - wx) + v01 * wx
99                bot = v10 * (1.0 - wx) + v11 * wx
100                out[i2][j2][c] = top * (1.0 - wy) + bot * wy
101 
102    return out
